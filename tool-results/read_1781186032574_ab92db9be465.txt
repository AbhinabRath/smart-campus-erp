    49→// Grade helper
    50→function getGrade(pct: number): { grade: string; color: string; bg: string } {
    51→  if (pct >= 90) return { grade: 'A+', color: 'text-emerald-700 dark:text-emerald-400', bg: 'bg-emerald-100 dark:bg-emerald-950/50' };
    52→  if (pct >= 80) return { grade: 'A', color: 'text-emerald-600 dark:text-emerald-400', bg: 'bg-emerald-50 dark:bg-emerald-950/30' };
    53→  if (pct >= 70) return { grade: 'B', color: 'text-sky-700 dark:text-sky-400', bg: 'bg-sky-50 dark:bg-sky-950/30' };
    54→  if (pct >= 60) return { grade: 'C', color: 'text-amber-700 dark:text-amber-400', bg: 'bg-amber-50 dark:bg-amber-950/30' };
    55→  if (pct >= 50) return { grade: 'D', color: 'text-orange-700 dark:text-orange-400', bg: 'bg-orange-50 dark:bg-orange-950/30' };
    56→  return { grade: 'F', color: 'text-red-700 dark:text-red-400', bg: 'bg-red-50 dark:bg-red-950/30' };
    57→}
