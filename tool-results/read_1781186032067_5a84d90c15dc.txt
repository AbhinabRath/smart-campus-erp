     1→// =============================================================================
     2→// Smart Campus ERP - Assignment Manager Component (Enhanced)
     3→// =============================================================================
     4→// Role-based assignment management:
     5→// - Teacher: Create assignments with deadline picker, view submissions, grade
     6→// - Student: View assignment cards with status badges, submit work, file upload UI
     7→// - Admin: Full overview with department/subject filters, stat cards,

    10→
    11→'use client';
    12→
    13→import React, { useEffect, useState, useCallback } from 'react';
    14→import { motion } from 'framer-motion';
    15→import {
    16→  FileUp, Plus, RefreshCw, Upload, Eye, CheckCircle, Clock,
    17→  AlertTriangle, Calendar, FileText, Send, Trash2, Users, TrendingUp,
    18→  ClipboardList, Download,
    19→} from 'lucide-react';
    20→import { Card, CardContent, CardHeader, CardTitle } from '@/components/ui/card';
    21→import { Button } from '@/components/ui/button';
    22→import { Input } from '@/components/ui/input';
    23→import { Label } from '@/components/ui/label';
    24→import { Textarea } from '@/components/ui/textarea';
    25→import { Badge } from '@/components/ui/badge';
    26→import { Skeleton } from '@/components/ui/skeleton';
    27→import { Select, SelectContent, SelectItem, SelectTrigger, SelectValue } from '@/components/ui/select';
    28→import { Dialog, DialogContent, DialogHeader, DialogTitle, DialogTrigger } from '@/components/ui/dialog';
    29→import { Popover, PopoverContent, PopoverTrigger } from '@/components/ui/popover';
    30→import { Calendar as CalendarComponent } from '@/components/ui/calendar';
    31→import { Table, TableBody, TableCell, TableHead, TableHeader, TableRow } from '@/components/ui/table';
    32→import { useAppStore } from '@/lib/store';
    33→import api from '@/lib/api';
    34→import { format } from 'date-fns';
    35→import { useToast } from '@/hooks/use-toast';
    36→
    37→interface Subject { id: string; name: string; code: string; departmentId?: string; department?: { name: string; code: string }; }
    38→interface Department { id: string; name: string; code: string; }
    39→interface Teacher { id: string; user: { name: string }; employeeId: string; }
    40→interface Assignment {
    41→  id: string; title: string; description: string; deadline: string; maxMarks: number; filePath: string | null; fileName: string | null;
    42→  subject: { id?: string; name: string; code: string; departmentId?: string };
    43→  teacher: { user: { name: string } };
    44→  submissions: Array<{
    45→    id: string; studentId: string; filePath: string | null; fileName: string | null;
    46→    submittedAt: string; status: string; marksObtained: number | null; feedback: string | null;
    47→    student: { user: { name: string }; rollNumber: string };
    48→  }>;
    49→  _count?: { submissions: number };
    50→}
    51→
    52→export default function AssignmentManager() {
    53→  const { currentUser } = useAppStore();
    54→  const role = currentUser?.role || 'student';
    55→  const { toast } = useToast();
    56→
    57→  const [assignments, setAssignments] = useState<Assignment[]>([]);
    58→  const [subjects, setSubjects] = useState<Subject[]>([]);
    59→  const [loading, setLoading] = useState(true);
    60→
    61→  // Create form state
    62→  const [title, setTitle] = useState('');
    63→  const [description, setDescription] = useState('');
    64→  const [deadline, setDeadline] = useState<Date | undefined>(undefined);
    65→  const [deadlineTime, setDeadlineTime] = useState('23:59');
    66→  const [maxMarks, setMaxMarks] = useState('100');
    67→  const [selectedSubject, setSelectedSubject] = useState('');
    68→  const [file, setFile] = useState<File | null>(null);
    69→  const [creating, setCreating] = useState(false);
    70→  const [createOpen, setCreateOpen] = useState(false);
    71→  const [calendarOpen, setCalendarOpen] = useState(false);
    72→
    73→  // Submission state - per assignment
    74→  const [submitFiles, setSubmitFiles] = useState<Record<string, File | null>>({});
    75→  const [submitting, setSubmitting] = useState<string | null>(null);
    76→
    77→  // Grading state
    78→  const [gradeId, setGradeId] = useState('');
    79→  const [gradeMarks, setGradeMarks] = useState('');
    80→  const [gradeFeedback, setGradeFeedback] = useState('');
    81→  const [grading, setGrading] = useState(false);
    82→  const [gradeOpen, setGradeOpen] = useState(false);
    83→
    84→  // View submissions dialog
    85→  const [viewAssignment, setViewAssignment] = useState<Assignment | null>(null);
    86→
    87→  // Admin view state
    88→  const [adminDepts, setAdminDepts] = useState<Department[]>([]);
    89→  const [adminFilterDept, setAdminFilterDept] = useState('all');
    90→  const [adminFilterSubject, setAdminFilterSubject] = useState('all');
    91→  const [adminTeachers, setAdminTeachers] = useState<Teacher[]>([]);
    92→  const [adminSelectedTeacher, setAdminSelectedTeacher] = useState('');
    93→  const [adminCreateOpen, setAdminCreateOpen] = useState(false);
    94→  const [adminCalendarOpen, setAdminCalendarOpen] = useState(false);
    95→  const [adminTitle, setAdminTitle] = useState('');
    96→  const [adminDescription, setAdminDescription] = useState('');
    97→  const [adminDeadline, setAdminDeadline] = useState<Date | undefined>(undefined);
    98→  const [adminDeadlineTime, setAdminDeadlineTime] = useState('23:59');
    99→  const [adminMaxMarks, setAdminMaxMarks] = useState('100');
   100→  const [adminSelectedSubject, setAdminSelectedSubject] = useState('');
   101→  const [adminCreating, setAdminCreating] = useState(false);
   102→
   103→  const loadData = useCallback(async () => {
   104→    setLoading(true);
   105→    try {
   106→      const [assignRes, subRes] = await Promise.all([
   107→        api.get('/assignments'),
   108→        api.get('/subjects'),
   109→      ]);
   110→      setAssignments(assignRes.data.data || []);
   111→      setSubjects(subRes.data.data || []);
   112→    } catch { /* ignore */ }
   113→    setLoading(false);
   114→  }, []);
   115→
   116→  // Admin data loading
   117→  const loadAdminData = useCallback(async () => {
   118→    setLoading(true);
   119→    try {
   120→      const [assignRes, subRes, deptRes, teacherRes] = await Promise.all([
   121→        api.get('/assignments'),
   122→        api.get('/subjects'),
   123→        api.get('/departments'),
   124→        api.get('/users?role=teacher'),
   125→      ]);
   126→      setAssignments(assignRes.data.data || []);
   127→      setSubjects(subRes.data.data || []);
   128→      setAdminDepts(deptRes.data.data || []);
   129→      // Transform teacher users to Teacher format
   130→      const teacherUsers = teacherRes.data.data || [];
   131→      const teacherList = teacherUsers.map((u: { id: string; name: string; teacher?: { id: string; employeeId: string } | null }) => ({
   132→        id: u.teacher?.id || u.id,
   133→        user: { name: u.name },
   134→        employeeId: u.teacher?.employeeId || 'N/A',
   135→      }));
   136→      setAdminTeachers(teacherList);
   137→    } catch { /* ignore */ }
   138→    setLoading(false);
   139→  }, []);
   140→
   141→  useEffect(() => {
   142→    if (role === 'admin') {
   143→      loadAdminData();
   144→    } else {
   145→      loadData();
   146→    }
   147→  }, [role, loadData, loadAdminData]);
   148→
   149→  // Teacher: Create assignment
   150→  const handleCreate = async (e: React.FormEvent) => {
   151→    e.preventDefault();
   152→    if (!title || !selectedSubject || !deadline) {
   153→      toast({ title: 'Error', description: 'Please fill title, subject, and deadline', variant: 'destructive' });
   154→      return;
   155→    }
   156→    setCreating(true);
   157→    try {
   158→      const deadlineDate = new Date(deadline);
   159→      const [hours, minutes] = deadlineTime.split(':').map(Number);
   160→      deadlineDate.setHours(hours || 23, minutes || 59, 0);
   161→
   162→      const formData = new FormData();
   163→      formData.append('title', title);
   164→      formData.append('description', description);
   165→      formData.append('subjectId', selectedSubject);
   166→      formData.append('deadline', deadlineDate.toISOString());
   167→      formData.append('maxMarks', maxMarks);
   168→      if (file) formData.append('file', file);
   169→
   170→      await api.post('/assignments', formData, { headers: { 'Content-Type': 'multipart/form-data' } });
   171→      toast({ title: 'Success', description: 'Assignment created' });
   172→      setCreateOpen(false);
   173→      setTitle(''); setDescription(''); setDeadline(undefined); setDeadlineTime('23:59'); setFile(null);
   174→      setSelectedSubject(''); setMaxMarks('100');
   175→      loadData();
   176→    } catch (err: unknown) {
   177→      const axiosErr = err as { response?: { data?: { message?: string } } };
   178→      toast({ title: 'Error', description: axiosErr.response?.data?.message || 'Failed to create assignment', variant: 'destructive' });
   179→    } finally {
   180→      setCreating(false);
   181→    }
   182→  };
   183→
   184→  // Admin: Create assignment on behalf of teacher
   185→  const handleAdminCreate = async (e: React.FormEvent) => {
   186→    e.preventDefault();
   187→    if (!adminTitle || !adminSelectedSubject || !adminDeadline || !adminSelectedTeacher) {
   188→      toast({ title: 'Error', description: 'Please fill title, subject, teacher, and deadline', variant: 'destructive' });
   189→      return;
   190→    }
   191→    setAdminCreating(true);
   192→    try {
   193→      const deadlineDate = new Date(adminDeadline);
   194→      const [hours, minutes] = adminDeadlineTime.split(':').map(Number);
   195→      deadlineDate.setHours(hours || 23, minutes || 59, 0);
   196→
   197→      await api.post('/assignments', {
   198→        title: adminTitle,
   199→        description: adminDescription,
   200→        subjectId: adminSelectedSubject,
   201→        deadline: deadlineDate.toISOString(),
   202→        maxMarks: adminMaxMarks,
   203→        teacherId: adminSelectedTeacher,
   204→      });
   205→      toast({ title: 'Success', description: 'Assignment created successfully' });
   206→      setAdminCreateOpen(false);
   207→      setAdminTitle(''); setAdminDescription(''); setAdminDeadline(undefined); setAdminDeadlineTime('23:59');
   208→      setAdminSelectedSubject(''); setAdminSelectedTeacher(''); setAdminMaxMarks('100');
   209→      loadAdminData();
   210→    } catch (err: unknown) {
   211→      const axiosErr = err as { response?: { data?: { message?: string } } };
   212→      toast({ title: 'Error', description: axiosErr.response?.data?.message || 'Failed to create assignment', variant: 'destructive' });
   213→    } finally {
   214→      setAdminCreating(false);
   215→    }
   216→  };
   217→
   218→  // Admin: Delete assignment
   219→  const handleAdminDelete = async (id: string) => {
   220→    try {
   221→      await api.delete(`/assignments/${id}`);
   222→      toast({ title: 'Deleted', description: 'Assignment deleted' });
   223→      loadAdminData();
   224→    } catch {
   225→      toast({ title: 'Error', description: 'Failed to delete assignment', variant: 'destructive' });
   226→    }
   227→  };
   228→
   229→  // Student: Submit assignment
   230→  const handleSubmit = async (assignmentId: string) => {
   231→    const submitFile = submitFiles[assignmentId];
   232→    if (!submitFile) {
   233→      toast({ title: 'Error', description: 'Please select a file', variant: 'destructive' });
   234→      return;
   235→    }
   236→    setSubmitting(assignmentId);
   237→    try {
   238→      const formData = new FormData();
   239→      formData.append('file', submitFile);
   240→      await api.post(`/assignments/${assignmentId}/submit`, formData, { headers: { 'Content-Type': 'multipart/form-data' } });
   241→      toast({ title: 'Success', description: 'Assignment submitted' });
   242→      setSubmitFiles((prev) => ({ ...prev, [assignmentId]: null }));
   243→      loadData();
   244→    } catch (err: unknown) {
   245→      const axiosErr = err as { response?: { data?: { message?: string } } };
   246→      toast({ title: 'Error', description: axiosErr.response?.data?.message || 'Failed to submit', variant: 'destructive' });
   247→    } finally {
   248→      setSubmitting(null);
   249→    }
   250→  };
   251→
   252→  // Teacher: Grade submission
   253→  const handleGrade = async () => {
   254→    if (!gradeId || !gradeMarks) return;
   255→    setGrading(true);
   256→    try {
   257→      await api.put(`/assignments/submissions/${gradeId}/grade`, {
   258→        marksObtained: parseFloat(gradeMarks),
   259→        feedback: gradeFeedback,
   260→      });
   261→      toast({ title: 'Success', description: 'Submission graded' });
   262→      setGradeOpen(false);
   263→      setGradeMarks(''); setGradeFeedback('');
   264→      loadData();
   265→    } catch {
   266→      toast({ title: 'Error', description: 'Failed to grade submission', variant: 'destructive' });
   267→    } finally {
   268→      setGrading(false);
   269→    }
   270→  };
   271→
   272→  const openGrade = (subId: string, currentMarks?: number | null, currentFeedback?: string | null) => {
   273→    setGradeId(subId);
   274→    setGradeMarks(currentMarks?.toString() || '');
   275→    setGradeFeedback(currentFeedback || '');
   276→    setGradeOpen(true);
   277→  };
   278→
   279→  const statusColor = (status: string) => {
   280→    switch (status) {
   281→      case 'graded': return 'bg-emerald-100 text-emerald-800 border-emerald-200';
   282→      case 'late': return 'bg-amber-100 text-amber-800 border-amber-200';
   283→      default: return 'bg-sky-100 text-sky-800 border-sky-200';
   284→    }
   285→  };
   286→
   287→  const deadlinePassed = (dl: string) => new Date(dl) < new Date();
   288→
   289→  // Get student assignment status
   290→  const getAssignmentStatus = (a: Assignment) => {
   291→    if (role !== 'student') return null;
   292→    const mySub = a.submissions?.[0];
   293→    if (mySub) {
   294→      return { type: 'submitted' as const, sub: mySub };
   295→    }
   296→    if (deadlinePassed(a.deadline)) {
   297→      return { type: 'overdue' as const, sub: null };
   298→    }
   299→    return { type: 'pending' as const, sub: null };
   300→  };
   301→
   302→  // Status badge for student view
   303→  const renderStatusBadge = (status: ReturnType<typeof getAssignmentStatus>) => {
   304→    if (!status) return null;
   305→    switch (status.type) {
   306→      case 'submitted':
   307→        return <Badge className="bg-emerald-100 text-emerald-800 border-emerald-200">
   308→          <CheckCircle className="w-3 h-3 mr-1" /> Submitted
   309→        </Badge>;
   310→      case 'overdue':
   311→        return <Badge variant="destructive">
   312→          <AlertTriangle className="w-3 h-3 mr-1" /> Overdue
   313→        </Badge>;
   314→      case 'pending':
   315→        return <Badge variant="outline" className="text-amber-700 border-amber-300 bg-amber-50">
   316→          <Clock className="w-3 h-3 mr-1" /> Pending
   317→        </Badge>;
   318→    }
   319→  };
   320→
   321→  // Admin: Export assignments CSV
   322→  const exportAdminCSV = () => {
   323→    const dataToExport = filteredAdminAssignments;
   324→    if (dataToExport.length === 0) {
   325→      toast({ title: 'No Data', description: 'No assignments to export', variant: 'destructive' });
   326→      return;
   327→    }
   328→    const headers = ['Title', 'Subject', 'Teacher', 'Due Date', 'Max Marks', 'Submissions', 'Status'];
   329→    const rows = dataToExport.map((a) => [
   330→      a.title,
   331→      a.subject?.name || 'N/A',
   332→      a.teacher?.user?.name || 'N/A',
   333→      format(new Date(a.deadline), 'yyyy-MM-dd h:mm a'),
   334→      a.maxMarks,
   335→      a._count?.submissions || 0,
   336→      deadlinePassed(a.deadline) ? 'Overdue' : 'Active',
   337→    ]);
   338→    const csv = [headers, ...rows].map((row) => row.map((cell) => `"${cell}"`).join(',')).join('\n');
   339→    const blob = new Blob([csv], { type: 'text/csv' });
   340→    const url = URL.createObjectURL(blob);
   341→    const a = document.createElement('a');
   342→    a.href = url;
   343→    a.download = `assignments-export-${format(new Date(), 'yyyy-MM-dd')}.csv`;
   344→    a.click();
   345→    URL.revokeObjectURL(url);
   346→    toast({ title: 'Exported', description: `${dataToExport.length} assignments exported successfully` });
   347→  };
   348→
   349→  // =========================================================================
   350→  // Admin View
   351→  // =========================================================================
   352→  if (role === 'admin') {
   353→    // Filter subjects by department
   354→    const filteredAdminSubjects = adminFilterDept === 'all'
   355→      ? subjects
   356→      : subjects.filter((s) => s.departmentId === adminFilterDept);
   357→
   358→    // Filter assignments
   359→    const filteredAdminAssignments = assignments.filter((a) => {
   360→      const deptMatch = adminFilterDept === 'all' || (a.subject?.id && subjects.find((s) => s.id === a.subject.id)?.departmentId === adminFilterDept);
   361→      const subMatch = adminFilterSubject === 'all' || a.subject?.id === adminFilterSubject;
   362→      return deptMatch && subMatch;
   363→    });
   364→
   365→    // Compute stats
   366→    const totalAssignments = filteredAdminAssignments.length;
   367→    const activeCount = filteredAdminAssignments.filter((a) => !deadlinePassed(a.deadline)).length;
   368→    const overdueCount = filteredAdminAssignments.filter((a) => deadlinePassed(a.deadline)).length;
   369→    const avgSubmissionRate = totalAssignments > 0
   370→      ? Math.round(
   371→          filteredAdminAssignments.reduce((sum, a) => {
   372→            const totalStudents = 6; // approximate from seed data
   373→            const subs = a._count?.submissions || 0;
   374→            return sum + (subs / totalStudents) * 100;
   375→          }, 0) / totalAssignments
   376→        )
   377→      : 0;
   378→
   379→    return (
   380→      <div className="space-y-6">
   381→        <div className="flex items-center justify-between">
   382→          <h2 className="text-2xl font-bold tracking-tight">Assignments Management</h2>
   383→          <div className="flex items-center gap-2">
   384→            <Button variant="outline" size="sm" onClick={exportAdminCSV} className="gap-1.5">
   385→              <Download className="w-4 h-4" /> Export
   386→            </Button>
   387→            <Dialog open={adminCreateOpen} onOpenChange={setAdminCreateOpen}>
   388→              <DialogTrigger asChild>
   389→                <Button className="bg-emerald-700 hover:bg-emerald-800"><Plus className="w-4 h-4 mr-2" /> Create Assignment</Button>
   390→              </DialogTrigger>
   391→              <DialogContent className="max-w-lg">
   392→                <DialogHeader><DialogTitle>Create Assignment (Admin)</DialogTitle></DialogHeader>
   393→                <form onSubmit={handleAdminCreate} className="space-y-4">
   394→                  <div className="space-y-2"><Label>Title *</Label><Input value={adminTitle} onChange={(e) => setAdminTitle(e.target.value)} required /></div>
   395→                  <div className="space-y-2"><Label>Description</Label><Textarea value={adminDescription} onChange={(e) => setAdminDescription(e.target.value)} rows={3} /></div>
   396→                  <div className="space-y-2">
   397→                    <Label>Teacher *</Label>
   398→                    <Select value={adminSelectedTeacher} onValueChange={setAdminSelectedTeacher}>
   399→                      <SelectTrigger><SelectValue placeholder="Select teacher" /></SelectTrigger>
   400→                      <SelectContent className="max-h-60">
   401→                        {adminTeachers.map((t) => <SelectItem key={t.id} value={t.id}>{t.user.name} ({t.employeeId})</SelectItem>)}
   402→                      </SelectContent>
   403→                    </Select>
   404→                  </div>
   405→                  <div className="space-y-2">
   406→                    <Label>Subject *</Label>
   407→                    <Select value={adminSelectedSubject} onValueChange={setAdminSelectedSubject}>
   408→                      <SelectTrigger><SelectValue placeholder="Select subject" /></SelectTrigger>
   409→                      <SelectContent>
   410→                        {subjects.map((s) => <SelectItem key={s.id} value={s.id}>{s.name} ({s.code})</SelectItem>)}
   411→                      </SelectContent>
   412→                    </Select>
   413→                  </div>
   414→                  <div className="grid grid-cols-2 gap-4">
   415→                    <div className="space-y-2">
   416→                      <Label>Deadline *</Label>
   417→                      <Popover open={adminCalendarOpen} onOpenChange={setAdminCalendarOpen}>
   418→                        <PopoverTrigger asChild>
   419→                          <Button variant="outline" className="w-full justify-start text-left font-normal">
   420→                            <Calendar className="w-4 h-4 mr-2" />
   421→                            {adminDeadline ? format(adminDeadline, 'PPP') : 'Pick a date'}
   422→                          </Button>
   423→                        </PopoverTrigger>
   424→                        <PopoverContent className="w-auto p-0" align="start">
   425→                          <CalendarComponent
   426→                            mode="single"
   427→                            selected={adminDeadline}
   428→                            onSelect={(d) => { setAdminDeadline(d); setAdminCalendarOpen(false); }}
   429→                          />
   430→                        </PopoverContent>
   431→                      </Popover>
   432→                    </div>
   433→                    <div className="space-y-2">
   434→                      <Label>Time</Label>
   435→                      <Input type="time" value={adminDeadlineTime} onChange={(e) => setAdminDeadlineTime(e.target.value)} />
   436→                    </div>
   437→                  </div>
   438→                  <div className="space-y-2">
   439→                    <Label>Max Marks</Label>
   440→                    <Input type="number" value={adminMaxMarks} onChange={(e) => setAdminMaxMarks(e.target.value)} />
   441→                  </div>
   442→                  <Button type="submit" disabled={adminCreating} className="w-full bg-emerald-700 hover:bg-emerald-800">
   443→                    {adminCreating ? 'Creating...' : 'Create Assignment'}
   444→                  </Button>
   445→                </form>
   446→              </DialogContent>
   447→            </Dialog>
   448→            <Button variant="ghost" size="icon" onClick={loadAdminData}><RefreshCw className="w-4 h-4" /></Button>
   449→          </div>
   450→        </div>
   451→
   452→        {/* Filters */}
   453→        <motion.div initial={{ opacity: 0, y: 20 }} animate={{ opacity: 1, y: 0 }}>
   454→          <Card>
   455→            <CardContent className="p-4">
   456→              <div className="flex flex-wrap items-center gap-4">
   457→                <div className="flex items-center gap-2">
   458→                  <Label className="text-sm whitespace-nowrap">Department:</Label>
   459→                  <Select value={adminFilterDept} onValueChange={(v) => { setAdminFilterDept(v); setAdminFilterSubject('all'); }}>
   460→                    <SelectTrigger className="w-[180px]"><SelectValue placeholder="All Departments" /></SelectTrigger>
   461→                    <SelectContent>
   462→                      <SelectItem value="all">All Departments</SelectItem>
   463→                      {adminDepts.map((d) => <SelectItem key={d.id} value={d.id}>{d.name}</SelectItem>)}
   464→                    </SelectContent>
   465→                  </Select>
   466→                </div>
   467→                <div className="flex items-center gap-2">
   468→                  <Label className="text-sm whitespace-nowrap">Subject:</Label>
   469→                  <Select value={adminFilterSubject} onValueChange={setAdminFilterSubject}>
   470→                    <SelectTrigger className="w-[200px]"><SelectValue placeholder="All Subjects" /></SelectTrigger>
   471→                    <SelectContent>
   472→                      <SelectItem value="all">All Subjects</SelectItem>
   473→                      {filteredAdminSubjects.map((s) => <SelectItem key={s.id} value={s.id}>{s.name} ({s.code})</SelectItem>)}
   474→                    </SelectContent>
   475→                  </Select>
   476→                </div>
   477→              </div>
   478→            </CardContent>
   479→          </Card>
   480→        </motion.div>
   481→
   482→        {/* Stat Cards */}
   483→        {loading ? (
   484→          <div className="grid grid-cols-2 md:grid-cols-4 gap-4">
   485→            {[...Array(4)].map((_, i) => <Skeleton key={i} className="h-28 rounded-xl" />)}
