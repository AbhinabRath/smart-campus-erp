     1→// =============================================================================
     2→// Smart Campus ERP - Attendance Manager Component
     3→// =============================================================================
     4→// Role-based attendance management:
     5→// - Teacher: Create sessions, display QR code, end sessions, view reports
     6→// - Student: Enter QR code to mark attendance, view history & percentage
     7→// - Admin: View attendance reports and analytics
     8→// =============================================================================
     9→
    10→'use client';
    11→
    12→import React, { useEffect, useState, useCallback } from 'react';
    13→import { motion, AnimatePresence } from 'framer-motion';
    14→import {
    15→  ClipboardCheck, QrCode, Play, StopCircle, RefreshCw, CheckCircle, XCircle, Users, Download, BarChart3
    16→} from 'lucide-react';
    17→import { Card, CardContent, CardHeader, CardTitle } from '@/components/ui/card';
    18→import { Button } from '@/components/ui/button';
    19→import { Input } from '@/components/ui/input';
    20→import { Label } from '@/components/ui/label';
    21→import { Badge } from '@/components/ui/badge';
    22→import { Select, SelectContent, SelectItem, SelectTrigger, SelectValue } from '@/components/ui/select';
    23→import { Table, TableBody, TableCell, TableHead, TableHeader, TableRow } from '@/components/ui/table';
    24→import { useAppStore } from '@/lib/store';
    25→import api from '@/lib/api';
    26→import { format } from 'date-fns';
    27→import { useToast } from '@/hooks/use-toast';
    28→
    29→interface Subject { id: string; name: string; code: string; }
    30→interface Department { id: string; name: string; code: string; }
    31→interface Session {
    32→  id: string; qrCode: string; isActive: boolean; duration: number; startedAt: string; endedAt: string | null;
    33→  subject: { name: string; code: string }; teacher: { user: { name: string } };
    34→  department: { name: string }; semester: number; section: string; _count: { records: number };
    35→  qrDataUrl?: string;
    36→}
    37→interface Record { id: string; markedAt: string; session: { subject: { name: string }; startedAt: string; }; }
    38→
    39→export default function AttendanceManager() {
    40→  const { currentUser } = useAppStore();
    41→  const role = currentUser?.role || 'student';
    42→  const { toast } = useToast();
    43→
    44→  // Teacher state
    45→  const [subjects, setSubjects] = useState<Subject[]>([]);
    46→  const [departments, setDepartments] = useState<Department[]>([]);
    47→  const [sessions, setSessions] = useState<Session[]>([]);
    48→  const [activeSession, setActiveSession] = useState<Session | null>(null);
    49→  const [selectedSubject, setSelectedSubject] = useState('');
    50→  const [selectedDept, setSelectedDept] = useState('');
    51→  const [selectedSemester, setSelectedSemester] = useState('3');
    52→  const [selectedSection, setSelectedSection] = useState('A');
    53→  const [creating, setCreating] = useState(false);
    54→
    55→  // Student state
    56→  const [qrInput, setQrInput] = useState('');
    57→  const [marking, setMarking] = useState(false);
    58→  const [myRecords, setMyRecords] = useState<Record[]>([]);
    59→  const [attendancePercent, setAttendancePercent] = useState({ percentage: 0, totalSessions: 0, attended: 0 });
    60→
    61→  // Admin state
    62→  const [adminFilterDept, setAdminFilterDept] = useState('all');
    63→  const [adminFilterSemester, setAdminFilterSemester] = useState('all');
    64→  const [adminLoading, setAdminLoading] = useState(true);
    65→
    66→  // Load dropdown data for teachers
    67→  const loadTeacherData = useCallback(async () => {
    68→    try {
    69→      const [subRes, deptRes, sessRes] = await Promise.all([
    70→        api.get('/subjects'),
    71→        api.get('/departments'),
    72→        api.get('/attendance/sessions'),
    73→      ]);
    74→      setSubjects(subRes.data.data || []);
    75→      setDepartments(deptRes.data.data || []);
    76→      const allSessions = sessRes.data.data || [];
    77→      setSessions(allSessions);
    78→      const active = allSessions.find((s: Session) => s.isActive);
    79→      if (active) setActiveSession(active);
    80→    } catch { /* ignore */ }
    81→  }, []);
    82→
    83→  // Load student data
    84→  const loadStudentData = useCallback(async () => {
    85→    try {
    86→      const [histRes, pctRes] = await Promise.all([
    87→        api.get('/attendance/my-attendance'),
    88→        api.get('/attendance/percentage'),
    89→      ]);
    90→      setMyRecords(histRes.data.data || []);
    91→      setAttendancePercent(pctRes.data.data || { percentage: 0, totalSessions: 0, attended: 0 });
    92→    } catch { /* ignore */ }
    93→  }, []);
    94→
    95→  // Admin: load data with loading state
    96→  const loadAdminData = useCallback(async () => {
    97→    setAdminLoading(true);
    98→    try {
    99→      const [sessRes, deptRes] = await Promise.all([
   100→        api.get('/attendance/sessions'),
   101→        api.get('/departments'),
   102→      ]);
   103→      const allSessions = sessRes.data.data || [];
   104→      setSessions(allSessions);
   105→      setDepartments(deptRes.data.data || []);
   106→    } catch { /* ignore */ }
   107→    setAdminLoading(false);
   108→  }, []);
   109→
   110→  useEffect(() => {
   111→    if (role === 'teacher') loadTeacherData();
   112→    if (role === 'admin') loadAdminData();
   113→    if (role === 'student') loadStudentData();
   114→  }, [role, loadTeacherData, loadStudentData, loadAdminData]);
   115→
   116→  // Teacher: Create attendance session
   117→  const createSession = async () => {
   118→    if (!selectedSubject || !selectedDept) {
   119→      toast({ title: 'Error', description: 'Please select subject and department', variant: 'destructive' });
   120→      return;
   121→    }
   122→    setCreating(true);
   123→    try {
   124→      const res = await api.post('/attendance/sessions', {
   125→        subjectId: selectedSubject,
   126→        departmentId: selectedDept,
   127→        semester: parseInt(selectedSemester),
   128→        section: selectedSection,
   129→      });
   130→      const newSession = res.data.data;
   131→      setActiveSession(newSession);
   132→      setSessions((prev) => [newSession, ...prev]);
   133→      toast({ title: 'Session Created', description: 'Attendance session is now active' });
   134→    } catch (err: unknown) {
   135→      const axiosErr = err as { response?: { data?: { message?: string } } };
   136→      toast({ title: 'Error', description: axiosErr.response?.data?.message || 'Failed to create session', variant: 'destructive' });
   137→    } finally {
   138→      setCreating(false);
   139→    }
   140→  };
   141→
   142→  // Teacher: End attendance session
   143→  const endSession = async (sessionId: string) => {
   144→    try {
   145→      await api.post(`/attendance/sessions/${sessionId}/end`);
   146→      setActiveSession(null);
   147→      setSessions((prev) => prev.map((s) => s.id === sessionId ? { ...s, isActive: false } : s));
   148→      toast({ title: 'Session Ended', description: 'Attendance session has been closed' });
   149→    } catch {
   150→      toast({ title: 'Error', description: 'Failed to end session', variant: 'destructive' });
   151→    }
   152→  };
   153→
   154→  // Export CSV helper
   155→  const exportSessionsCSV = (sessionsData: Session[], filename: string) => {
   156→    if (sessionsData.length === 0) {
   157→      toast({ title: 'No Data', description: 'No sessions to export', variant: 'destructive' });
   158→      return;
   159→    }
   160→    const headers = ['Subject', 'Teacher', 'Department', 'Semester', 'Section', 'Started', 'Present Count', 'Status'];
   161→    const rows = sessionsData.map((s) => [
   162→      s.subject?.name || 'N/A',
   163→      s.teacher?.user?.name || 'N/A',
   164→      s.department?.name || 'N/A',
   165→      s.semester?.toString() || '',
   166→      s.section || '',
   167→      s.startedAt ? format(new Date(s.startedAt), 'MMM d, yyyy h:mm a') : '',
   168→      (s._count?.records || 0).toString(),
   169→      s.isActive ? 'Live' : 'Ended',
   170→    ]);
   171→    const csvContent = [headers, ...rows].map((row) => row.map((cell) => `"${cell.replace(/"/g, '""')}"`).join(',')).join('\n');
   172→    const blob = new Blob([csvContent], { type: 'text/csv;charset=utf-8;' });
   173→    const url = URL.createObjectURL(blob);
   174→    const link = document.createElement('a');
   175→    link.href = url;
   176→    link.download = filename;
   177→    link.click();
   178→    URL.revokeObjectURL(url);
   179→    toast({ title: 'Exported', description: 'CSV file downloaded successfully' });
   180→  };
   181→
   182→  // Student: Mark attendance via QR code
   183→  const markAttendance = async () => {
   184→    if (!qrInput.trim()) return;
   185→    setMarking(true);
   186→    try {
   187→      await api.post('/attendance/mark', { qrCode: qrInput.trim() });
   188→      toast({ title: 'Success', description: 'Attendance marked successfully!' });
   189→      setQrInput('');
   190→      loadStudentData();
   191→    } catch (err: unknown) {
   192→      const axiosErr = err as { response?: { data?: { message?: string } } };
   193→      toast({ title: 'Error', description: axiosErr.response?.data?.message || 'Failed to mark attendance', variant: 'destructive' });
   194→    } finally {
   195→      setMarking(false);
   196→    }
   197→  };
   198→
   199→  // =========================================================================
   200→  // Teacher View
   201→  // =========================================================================
   202→  if (role === 'teacher') {
   203→    return (
   204→      <div className="space-y-6">
   205→        <h2 className="text-2xl font-bold tracking-tight">Attendance Management</h2>
   206→
   207→        <div className="grid grid-cols-1 lg:grid-cols-2 gap-6">
   208→          {/* Create Session */}
   209→          <motion.div initial={{ opacity: 0, y: 20 }} animate={{ opacity: 1, y: 0 }}>
   210→            <Card>
   211→              <CardHeader><CardTitle className="text-lg">Create Session</CardTitle></CardHeader>
   212→              <CardContent className="space-y-4">
   213→                <div className="space-y-2">
   214→                  <Label>Subject</Label>
   215→                  <Select value={selectedSubject} onValueChange={setSelectedSubject}>
   216→                    <SelectTrigger><SelectValue placeholder="Select subject" /></SelectTrigger>
   217→                    <SelectContent>
   218→                      {subjects.map((s) => <SelectItem key={s.id} value={s.id}>{s.name} ({s.code})</SelectItem>)}
   219→                    </SelectContent>
   220→                  </Select>
   221→                </div>
   222→                <div className="space-y-2">
   223→                  <Label>Department</Label>
   224→                  <Select value={selectedDept} onValueChange={setSelectedDept}>
   225→                    <SelectTrigger><SelectValue placeholder="Select department" /></SelectTrigger>
   226→                    <SelectContent>
   227→                      {departments.map((d) => <SelectItem key={d.id} value={d.id}>{d.name} ({d.code})</SelectItem>)}
   228→                    </SelectContent>
   229→                  </Select>
   230→                </div>
   231→                <div className="grid grid-cols-2 gap-4">
   232→                  <div className="space-y-2">
   233→                    <Label>Semester</Label>
   234→                    <Select value={selectedSemester} onValueChange={setSelectedSemester}>
   235→                      <SelectTrigger><SelectValue /></SelectTrigger>
   236→                      <SelectContent>
   237→                        {[1,2,3,4,5,6,7,8].map((s) => <SelectItem key={s} value={s.toString()}>{s}</SelectItem>)}
   238→                      </SelectContent>
   239→                    </Select>
   240→                  </div>
   241→                  <div className="space-y-2">
   242→                    <Label>Section</Label>
   243→                    <Select value={selectedSection} onValueChange={setSelectedSection}>
   244→                      <SelectTrigger><SelectValue /></SelectTrigger>
   245→                      <SelectContent>
   246→                        {['A','B','C'].map((s) => <SelectItem key={s} value={s}>{s}</SelectItem>)}
   247→                      </SelectContent>
   248→                    </Select>
   249→                  </div>
   250→                </div>
   251→                <Button onClick={createSession} disabled={creating || !!activeSession} className="w-full bg-emerald-700 hover:bg-emerald-800">
   252→                  {creating ? 'Creating...' : activeSession ? 'Session Active' : 'Create Session'}
   253→                </Button>
   254→              </CardContent>
   255→            </Card>
   256→          </motion.div>
   257→
   258→          {/* Active Session / QR Code */}
   259→          <motion.div initial={{ opacity: 0, y: 20 }} animate={{ opacity: 1, y: 0 }} transition={{ delay: 0.1 }}>
   260→            <Card>
   261→              <CardHeader><CardTitle className="text-lg">Active Session</CardTitle></CardHeader>
   262→              <CardContent>
   263→                {activeSession ? (
   264→                  <div className="space-y-4 text-center">
   265→                    <Badge className="bg-emerald-600 mb-2">Live</Badge>
   266→                    <p className="font-medium">{activeSession.subject?.name || 'Session'}</p>
   267→                    <p className="text-sm text-muted-foreground">
   268→                      Started {format(new Date(activeSession.startedAt), 'h:mm a')}
   269→                    </p>
   270→                    {activeSession.qrDataUrl && (
   271→                      <div className="flex justify-center">
   272→                        <img src={activeSession.qrDataUrl} alt="QR Code" className="w-48 h-48 rounded-lg border" />
   273→                      </div>
   274→                    )}
   275→                    <p className="text-xs text-muted-foreground">QR Token: {activeSession.qrCode}</p>
   276→                    <Button variant="destructive" onClick={() => endSession(activeSession.id)} className="w-full">
   277→                      <StopCircle className="w-4 h-4 mr-2" /> End Session
   278→                    </Button>
   279→                  </div>
   280→                ) : (
   281→                  <div className="text-center py-8 text-muted-foreground">
   282→                    <QrCode className="w-12 h-12 mx-auto mb-3 opacity-30" />
   283→                    <p>No active session</p>
   284→                    <p className="text-sm">Create a session to generate a QR code</p>
   285→                  </div>
   286→                )}
   287→              </CardContent>
   288→            </Card>
   289→          </motion.div>
   290→        </div>
   291→
   292→        {/* Session History */}
   293→        <motion.div initial={{ opacity: 0, y: 20 }} animate={{ opacity: 1, y: 0 }} transition={{ delay: 0.2 }}>
   294→          <Card>
   295→            <CardHeader className="flex flex-row items-center justify-between">
   296→              <CardTitle className="text-lg">Session History</CardTitle>
   297→              <div className="flex items-center gap-1">
   298→                <Button variant="ghost" size="icon" onClick={() => exportSessionsCSV(sessions, 'attendance-sessions.csv')} title="Export CSV">
   299→                  <Download className="w-4 h-4" />
   300→                </Button>
   301→                <Button variant="ghost" size="icon" onClick={loadTeacherData}><RefreshCw className="w-4 h-4" /></Button>
   302→              </div>
   303→            </CardHeader>
   304→            <CardContent className="max-h-96 overflow-y-auto custom-scrollbar">
   305→              {sessions.length === 0 ? (
   306→                <p className="text-center text-muted-foreground py-4">No sessions yet</p>
   307→              ) : (
   308→                <Table>
   309→                  <TableHeader>
   310→                    <TableRow>
   311→                      <TableHead>Subject</TableHead>
   312→                      <TableHead>Sem/Sec</TableHead>
   313→                      <TableHead>Started</TableHead>
   314→                      <TableHead>Present</TableHead>
   315→                      <TableHead>Status</TableHead>
   316→                      <TableHead>Actions</TableHead>
   317→                    </TableRow>
   318→                  </TableHeader>
   319→                  <TableBody>
   320→                    {sessions.map((s) => (
   321→                      <TableRow key={s.id}>
   322→                        <TableCell className="font-medium">{s.subject?.name || 'N/A'}</TableCell>
   323→                        <TableCell>{s.semester}/{s.section}</TableCell>
   324→                        <TableCell>{format(new Date(s.startedAt), 'MMM d, h:mm a')}</TableCell>
   325→                        <TableCell>{s._count?.records || 0}</TableCell>
   326→                        <TableCell>
   327→                          <Badge variant={s.isActive ? 'default' : 'secondary'}>{s.isActive ? 'Live' : 'Ended'}</Badge>
   328→                        </TableCell>
   329→                        <TableCell>
   330→                          {s.isActive && (
   331→                            <Button size="sm" variant="outline" onClick={() => endSession(s.id)}>End</Button>
   332→                          )}
   333→                        </TableCell>
   334→                      </TableRow>
   335→                    ))}
   336→                  </TableBody>
   337→                </Table>
   338→              )}
   339→            </CardContent>
   340→          </Card>
   341→        </motion.div>
   342→      </div>
   343→    );
   344→  }
   345→
   346→  // =========================================================================
   347→  // Student View
   348→  // =========================================================================
   349→  if (role === 'student') {
   350→    return (
   351→      <div className="space-y-6">
   352→        <h2 className="text-2xl font-bold tracking-tight">My Attendance</h2>
   353→
   354→        <div className="grid grid-cols-1 sm:grid-cols-3 gap-4">
   355→          <motion.div initial={{ opacity: 0, y: 20 }} animate={{ opacity: 1, y: 0 }}>
   356→            <Card>
   357→              <CardContent className="p-5 text-center">
   358→                <p className="text-sm text-muted-foreground">Attendance %</p>
   359→                <p className={`text-3xl font-bold ${attendancePercent.percentage >= 75 ? 'text-emerald-600' : 'text-amber-600'}`}>
   360→                  {attendancePercent.percentage}%
   361→                </p>
   362→              </CardContent>
   363→            </Card>
   364→          </motion.div>
   365→          <motion.div initial={{ opacity: 0, y: 20 }} animate={{ opacity: 1, y: 0 }} transition={{ delay: 0.1 }}>
   366→            <Card>
   367→              <CardContent className="p-5 text-center">
   368→                <p className="text-sm text-muted-foreground">Total Sessions</p>
   369→                <p className="text-3xl font-bold">{attendancePercent.totalSessions}</p>
   370→              </CardContent>
   371→            </Card>
   372→          </motion.div>
   373→          <motion.div initial={{ opacity: 0, y: 20 }} animate={{ opacity: 1, y: 0 }} transition={{ delay: 0.2 }}>
   374→            <Card>
   375→              <CardContent className="p-5 text-center">
   376→                <p className="text-sm text-muted-foreground">Attended</p>
   377→                <p className="text-3xl font-bold text-emerald-600">{attendancePercent.attended}</p>
   378→              </CardContent>
   379→            </Card>
   380→          </motion.div>
   381→        </div>
   382→
   383→        {/* QR Code Input */}
   384→        <motion.div initial={{ opacity: 0, y: 20 }} animate={{ opacity: 1, y: 0 }} transition={{ delay: 0.3 }}>
   385→          <Card>
   386→            <CardHeader><CardTitle className="text-lg">Mark Attendance</CardTitle></CardHeader>
   387→            <CardContent>
   388→              <div className="flex gap-3">
   389→                <Input
   390→                  placeholder="Enter QR code token..."
   391→                  value={qrInput}
   392→                  onChange={(e) => setQrInput(e.target.value)}
   393→                  onKeyDown={(e) => e.key === 'Enter' && markAttendance()}
   394→                  className="flex-1"
   395→                />
   396→                <Button onClick={markAttendance} disabled={marking || !qrInput.trim()} className="bg-emerald-700 hover:bg-emerald-800">
   397→                  {marking ? 'Marking...' : <><CheckCircle className="w-4 h-4 mr-2" /> Mark</>}
   398→                </Button>
   399→              </div>
   400→              <p className="text-xs text-muted-foreground mt-2">Enter the QR code token provided by your teacher to mark attendance</p>
   401→            </CardContent>
   402→          </Card>
   403→        </motion.div>
   404→
   405→        {/* Attendance History */}
   406→        <motion.div initial={{ opacity: 0, y: 20 }} animate={{ opacity: 1, y: 0 }} transition={{ delay: 0.4 }}>
   407→          <Card>
   408→            <CardHeader className="flex flex-row items-center justify-between">
   409→              <CardTitle className="text-lg">Attendance History</CardTitle>
   410→              <Button variant="ghost" size="icon" onClick={loadStudentData}><RefreshCw className="w-4 h-4" /></Button>
   411→            </CardHeader>
   412→            <CardContent className="max-h-96 overflow-y-auto custom-scrollbar">
   413→              {myRecords.length === 0 ? (
   414→                <div className="text-center py-8 text-muted-foreground">
   415→                  <ClipboardCheck className="w-12 h-12 mx-auto mb-3 opacity-30" />
   416→                  <p>No attendance records yet</p>
   417→                </div>
   418→              ) : (
   419→                <Table>
   420→                  <TableHeader>
   421→                    <TableRow>
   422→                      <TableHead>Subject</TableHead>
   423→                      <TableHead>Date</TableHead>
   424→                      <TableHead>Status</TableHead>
   425→                    </TableRow>
   426→                  </TableHeader>
   427→                  <TableBody>
   428→                    {myRecords.map((r) => (
   429→                      <TableRow key={r.id}>
   430→                        <TableCell className="font-medium">{r.session?.subject?.name || 'N/A'}</TableCell>
   431→                        <TableCell>{format(new Date(r.markedAt), 'MMM d, h:mm a')}</TableCell>
   432→                        <TableCell><Badge className="bg-emerald-600">Present</Badge></TableCell>
   433→                      </TableRow>
   434→                    ))}
   435→                  </TableBody>
   436→                </Table>
   437→              )}
   438→            </CardContent>
   439→          </Card>
   440→        </motion.div>
   441→      </div>
   442→    );
   443→  }
   444→
   445→  // =========================================================================
   446→  // Admin View
   447→  // =========================================================================
   448→
   449→  // Admin filter logic
   450→  const adminFilteredSessions = sessions.filter((s) => {
   451→    if (adminFilterDept !== 'all' && s.department?.id !== adminFilterDept) return false;
   452→    if (adminFilterSemester !== 'all' && s.semester?.toString() !== adminFilterSemester) return false;
   453→    return true;
   454→  });
   455→
   456→  // Admin stats
   457→  const totalSessions = adminFilteredSessions.length;
   458→  const activeSessions = adminFilteredSessions.filter((s) => s.isActive).length;
   459→  const totalPresent = adminFilteredSessions.reduce((sum, s) => sum + (s._count?.records || 0), 0);
   460→  const avgAttendance = totalSessions > 0 ? Math.round((totalPresent / totalSessions) * 10) / 10 : 0;
   461→
   462→  return (
   463→    <div className="space-y-6">
   464→      <h2 className="text-2xl font-bold tracking-tight">Attendance Overview</h2>
   465→
   466→      {/* Stat Cards */}
   467→      <div className="grid grid-cols-1 sm:grid-cols-3 gap-4">
   468→        <motion.div initial={{ opacity: 0, y: 20 }} animate={{ opacity: 1, y: 0 }}>
   469→          <Card>
   470→            <CardContent className="p-5 text-center">
   471→              <p className="text-sm text-muted-foreground">Total Sessions</p>
   472→              <p className="text-3xl font-bold">{totalSessions}</p>
   473→            </CardContent>
   474→          </Card>
   475→        </motion.div>
   476→        <motion.div initial={{ opacity: 0, y: 20 }} animate={{ opacity: 1, y: 0 }} transition={{ delay: 0.1 }}>
   477→          <Card>
   478→            <CardContent className="p-5 text-center">
   479→              <p className="text-sm text-muted-foreground">Active Sessions</p>
   480→              <p className="text-3xl font-bold text-emerald-600">{activeSessions}</p>
   481→            </CardContent>
   482→          </Card>
   483→        </motion.div>
   484→        <motion.div initial={{ opacity: 0, y: 20 }} animate={{ opacity: 1, y: 0 }} transition={{ delay: 0.2 }}>
   485→          <Card>
   486→            <CardContent className="p-5 text-center">
   487→              <p className="text-sm text-muted-foreground">Avg Attendance</p>
   488→              <p className="text-3xl font-bold text-amber-600">{avgAttendance}</p>
   489→            </CardContent>
   490→          </Card>
