     4→// Comprehensive settings page organized into tabs: General (appearance/theme),

     9→'use client';
    10→
    11→import React, { useState, useEffect, useCallback } from 'react';
    12→import { motion } from 'framer-motion';
    13→import { useTheme } from 'next-themes';
    14→import {
    15→  Settings, Sun, Moon, Monitor, Bell, BellOff, Mail, ClipboardCheck,
    16→  FileText, AlertCircle, Lightbulb, User, Shield, KeyRound, Eye, EyeOff,
    17→  Trash2, AlertTriangle, ChevronRight, Type, PanelLeftClose, Check,
    18→} from 'lucide-react';
    19→import { Card, CardContent, CardHeader, CardTitle, CardDescription } from '@/components/ui/card';
    20→import { Button } from '@/components/ui/button';
    21→import { Input } from '@/components/ui/input';
    22→import { Label } from '@/components/ui/label';
    23→import { Switch } from '@/components/ui/switch';
    24→import { Separator } from '@/components/ui/separator';
    25→import { Badge } from '@/components/ui/badge';
    26→import {
    27→  Select, SelectContent, SelectItem, SelectTrigger, SelectValue,
    28→} from '@/components/ui/select';
    29→import {
    30→  Tabs, TabsContent, TabsList, TabsTrigger,
    31→} from '@/components/ui/tabs';
    32→import {
    33→  Tooltip, TooltipContent, TooltipProvider, TooltipTrigger,
    34→} from '@/components/ui/tooltip';
    35→import { useAppStore } from '@/lib/store';
    36→import api from '@/lib/api';
    37→import { useToast } from '@/hooks/use-toast';
    38→
    39→// ---------------------------------------------------------------------------
    40→// localStorage helpers with campus-erp- prefix
    41→// ---------------------------------------------------------------------------
    42→
    43→const LS_NOTIFICATIONS = 'campus-erp-notifications';
    44→const LS_SIDEBAR_COMPACT = 'campus-erp-sidebar-compact';
    45→const LS_FONT_SIZE = 'campus-erp-font-size';
    46→
    47→interface NotificationPrefs {
    48→  email: boolean;
    49→  attendance: boolean;
    50→  assignments: boolean;
    51→  notices: boolean;
    52→  recommendations: boolean;
    53→}
    54→
    55→const defaultNotificationPrefs: NotificationPrefs = {
    56→  email: true,
    57→  attendance: true,
    58→  assignments: true,
    59→  notices: true,
    60→  recommendations: true,
    61→};
    62→
    63→function loadNotificationPrefs(): NotificationPrefs {
    64→  if (typeof window === 'undefined') return defaultNotificationPrefs;
    65→  try {
    66→    const raw = localStorage.getItem(LS_NOTIFICATIONS);
    67→    if (raw) return { ...defaultNotificationPrefs, ...JSON.parse(raw) };
    68→  } catch { /* ignore */ }
    69→  return defaultNotificationPrefs;
    70→}
    71→
    72→function saveNotificationPrefs(prefs: NotificationPrefs) {
    73→  try { localStorage.setItem(LS_NOTIFICATIONS, JSON.stringify(prefs)); } catch { /* ignore */ }
    74→}
    75→
    76→function loadSidebarCompact(): boolean {
    77→  if (typeof window === 'undefined') return false;
    78→  return localStorage.getItem(LS_SIDEBAR_COMPACT) === 'true';
    79→}
    80→
    81→function saveSidebarCompact(val: boolean) {
    82→  try { localStorage.setItem(LS_SIDEBAR_COMPACT, String(val)); } catch { /* ignore */ }
    83→}
    84→
    85→function loadFontSize(): string {
    86→  if (typeof window === 'undefined') return 'medium';
    87→  return localStorage.getItem(LS_FONT_SIZE) || 'medium';
    88→}
    89→
    90→function saveFontSize(val: string) {
    91→  try { localStorage.setItem(LS_FONT_SIZE, val); } catch { /* ignore */ }
    92→}
    93→
    94→// ---------------------------------------------------------------------------
    95→// Animation variants
    96→// ---------------------------------------------------------------------------
    97→
    98→const containerVariants = {
    99→  hidden: { opacity: 0 },
   100→  show: { opacity: 1, transition: { staggerChildren: 0.06 } },
   101→};
   102→
   103→const itemVariants = {
   104→  hidden: { opacity: 0, y: 12 },
   105→  show: { opacity: 1, y: 0, transition: { duration: 0.3 } },
   106→};
   107→
   108→// ---------------------------------------------------------------------------
   109→// SettingsPage Component
   110→// ---------------------------------------------------------------------------
   111→
   112→export default function SettingsPage() {
   113→  const { currentUser } = useAppStore();
   114→  const { theme, setTheme } = useTheme();
   115→  const { toast } = useToast();
   116→
   117→  // ---- General Tab State ----
   118→  const [sidebarCompact, setSidebarCompact] = useState(false);
   119→  const [fontSize, setFontSize] = useState('medium');
   120→
   121→  // ---- Notifications Tab State ----
   122→  const [notifPrefs, setNotifPrefs] = useState<NotificationPrefs>(defaultNotificationPrefs);
   123→
   124→  // ---- Security Tab State ----
   125→  const [currentPassword, setCurrentPassword] = useState('');
   126→  const [newPassword, setNewPassword] = useState('');
   127→  const [confirmPassword, setConfirmPassword] = useState('');
   128→  const [showCurrent, setShowCurrent] = useState(false);
   129→  const [showNew, setShowNew] = useState(false);
   130→  const [changingPassword, setChangingPassword] = useState(false);
   131→
   132→  // ---- Load localStorage on mount ----
   133→  useEffect(() => {
   134→    setSidebarCompact(loadSidebarCompact());
   135→    setFontSize(loadFontSize());
   136→    setNotifPrefs(loadNotificationPrefs());
   137→  }, []);
   138→
   139→  // ---- Handlers ----
   140→
   141→  const handleSidebarCompact = useCallback((val: boolean) => {
   142→    setSidebarCompact(val);
   143→    saveSidebarCompact(val);
   144→    toast({
   145→      title: val ? 'Compact sidebar enabled' : 'Compact sidebar disabled',
   146→      description: 'This preference is saved locally.',
   147→    });
   148→  }, [toast]);
   149→
   150→  const handleFontSize = useCallback((val: string) => {
   151→    setFontSize(val);
   152→    saveFontSize(val);
   153→    toast({
   154→      title: 'Font size updated',
   155→      description: `Font size set to ${val}.`,
   156→    });
   157→  }, [toast]);
   158→
   159→  const handleNotifToggle = useCallback((key: keyof NotificationPrefs, val: boolean) => {
   160→    setNotifPrefs((prev) => {
   161→      const updated = { ...prev, [key]: val };
   162→      saveNotificationPrefs(updated);
   163→      return updated;
   164→    });
   165→  }, []);
   166→
   167→  const handleChangePassword = async (e: React.FormEvent) => {
   168→    e.preventDefault();
   169→    if (!currentUser) return;
   170→
   171→    if (!currentPassword || !newPassword || !confirmPassword) {
   172→      toast({ title: 'Error', description: 'Please fill all password fields', variant: 'destructive' });
   173→      return;
   174→    }
   175→    if (newPassword.length < 6) {
   176→      toast({ title: 'Error', description: 'New password must be at least 6 characters', variant: 'destructive' });
   177→      return;
   178→    }
   179→    if (newPassword !== confirmPassword) {
   180→      toast({ title: 'Error', description: 'New passwords do not match', variant: 'destructive' });
   181→      return;
   182→    }
   183→
   184→    setChangingPassword(true);
   185→    try {
   186→      await api.put(`/users/${currentUser.id}/password`, {
   187→        currentPassword,
   188→        newPassword,
   189→      });
   190→      toast({ title: 'Success', description: 'Password changed successfully' });
   191→      setCurrentPassword('');
   192→      setNewPassword('');
   193→      setConfirmPassword('');
   194→    } catch (err: unknown) {
   195→      const axiosErr = err as { response?: { data?: { message?: string } } };
   196→      toast({
   197→        title: 'Error',
   198→        description: axiosErr.response?.data?.message || 'Failed to change password',
   199→        variant: 'destructive',
   200→      });
   201→    } finally {
   202→      setChangingPassword(false);
   203→    }
   204→  };
   205→
   206→  const navigateToProfile = () => {
   207→    useAppStore.getState().setView('profile');
   208→  };
   209→
   210→  // ---- Theme button helper ----
   211→  const themeOptions = [
   212→    { value: 'light', label: 'Light', icon: Sun },
   213→    { value: 'dark', label: 'Dark', icon: Moon },
   214→    { value: 'system', label: 'System', icon: Monitor },
   215→  ] as const;
   216→
   217→  if (!currentUser) {
   218→    return null;
   219→  }
   220→
   221→  return (
   222→    <motion.div variants={containerVariants} initial="hidden" animate="show" className="space-y-6">
   223→      {/* Page Title */}
   224→      <motion.div variants={itemVariants} className="flex items-center gap-3">
   225→        <div className="w-10 h-10 rounded-xl bg-emerald-100 dark:bg-emerald-900/30 flex items-center justify-center">
   226→          <Settings className="w-5 h-5 text-emerald-600 dark:text-emerald-400" />
   227→        </div>
   228→        <div>
   229→          <h2 className="text-2xl font-bold tracking-tight">Settings</h2>
   230→          <p className="text-sm text-muted-foreground">Manage your preferences and account security</p>
   231→        </div>
   232→      </motion.div>
   233→
   234→      {/* Tabs */}
   235→      <Tabs defaultValue="general" className="space-y-6">
   236→        <motion.div variants={itemVariants}>
   237→          <TabsList className="grid w-full max-w-md grid-cols-3">
   238→            <TabsTrigger value="general" className="gap-2">
   239→              <Settings className="w-4 h-4 hidden sm:block" />
   240→              General
   241→            </TabsTrigger>
   242→            <TabsTrigger value="notifications" className="gap-2">
   243→              <Bell className="w-4 h-4 hidden sm:block" />
   244→              Notifications
   245→            </TabsTrigger>
   246→            <TabsTrigger value="security" className="gap-2">
   247→              <Shield className="w-4 h-4 hidden sm:block" />
   248→              Security
   249→            </TabsTrigger>
   250→          </TabsList>
   251→        </motion.div>
   252→
   253→        {/* ================================================================= */}
   254→        {/* GENERAL TAB                                                       */}
   255→        {/* ================================================================= */}
   256→        <TabsContent value="general" className="space-y-6">
   257→          {/* Appearance Card */}
   258→          <motion.div variants={itemVariants}>
   259→            <Card>
   260→              <CardHeader>
   261→                <CardTitle className="text-lg flex items-center gap-2">
   262→                  <Sun className="w-5 h-5 text-amber-500" />
   263→                  Appearance
   264→                </CardTitle>
   265→                <CardDescription>Customize how the application looks and feels</CardDescription>
   266→              </CardHeader>
   267→              <CardContent className="space-y-6">
   268→                {/* Theme toggle */}
   269→                <div className="space-y-3">
   270→                  <Label className="text-sm font-medium">Theme</Label>
   271→                  <div className="grid grid-cols-3 gap-3 max-w-sm">
   272→                    {themeOptions.map((opt) => {
   273→                      const Icon = opt.icon;
   274→                      const isActive = theme === opt.value;
   275→                      return (
   276→                        <button
   277→                          key={opt.value}
   278→                          type="button"
   279→                          onClick={() => setTheme(opt.value)}
   280→                          className={`relative flex flex-col items-center gap-2 rounded-xl border-2 p-4 transition-all duration-200 hover:border-emerald-300 dark:hover:border-emerald-700 ${
   281→                            isActive
   282→                              ? 'border-emerald-500 bg-emerald-50 dark:bg-emerald-950/30 shadow-sm'
   283→                              : 'border-border bg-card'
   284→                          }`}
   285→                        >
   286→                          <Icon className={`w-6 h-6 ${isActive ? 'text-emerald-600 dark:text-emerald-400' : 'text-muted-foreground'}`} />
   287→                          <span className={`text-xs font-medium ${isActive ? 'text-emerald-700 dark:text-emerald-300' : 'text-muted-foreground'}`}>
   288→                            {opt.label}
   289→                          </span>
   290→                          {isActive && (
   291→                            <motion.div
   292→                              layoutId="theme-check"
   293→                              className="absolute top-2 right-2 w-5 h-5 rounded-full bg-emerald-500 flex items-center justify-center"
   294→                              transition={{ type: 'spring', stiffness: 350, damping: 25 }}
   295→                            >
   296→                              <Check className="w-3 h-3 text-white" />
   297→                            </motion.div>
   298→                          )}
   299→                        </button>
   300→                      );
   301→                    })}
   302→                  </div>
   303→                </div>
   304→
   305→                <Separator />
   306→
   307→                {/* Compact sidebar */}
   308→                <div className="flex items-center justify-between">
   309→                  <div className="flex items-center gap-3">
   310→                    <div className="w-9 h-9 rounded-lg bg-violet-100 dark:bg-violet-900/30 flex items-center justify-center">
   311→                      <PanelLeftClose className="w-4 h-4 text-violet-600 dark:text-violet-400" />
   312→                    </div>
   313→                    <div>
   314→                      <Label className="text-sm font-medium">Compact Sidebar</Label>
   315→                      <p className="text-xs text-muted-foreground">Use a narrower sidebar with icons only</p>
   316→                    </div>
   317→                  </div>
   318→                  <Switch
   319→                    checked={sidebarCompact}
   320→                    onCheckedChange={handleSidebarCompact}
   321→                  />
   322→                </div>
   323→
   324→                <Separator />
   325→
   326→                {/* Font size */}
   327→                <div className="flex items-center justify-between">
   328→                  <div className="flex items-center gap-3">
   329→                    <div className="w-9 h-9 rounded-lg bg-sky-100 dark:bg-sky-900/30 flex items-center justify-center">
   330→                      <Type className="w-4 h-4 text-sky-600 dark:text-sky-400" />
   331→                    </div>
   332→                    <div>
   333→                      <Label className="text-sm font-medium">Font Size</Label>
   334→                      <p className="text-xs text-muted-foreground">Adjust the base text size</p>
   335→                    </div>
   336→                  </div>
   337→                  <Select value={fontSize} onValueChange={handleFontSize}>
   338→                    <SelectTrigger className="w-32">
   339→                      <SelectValue />
   340→                    </SelectTrigger>
   341→                    <SelectContent>
   342→                      <SelectItem value="small">Small</SelectItem>
   343→                      <SelectItem value="medium">Medium</SelectItem>
   344→                      <SelectItem value="large">Large</SelectItem>
   345→                    </SelectContent>
   346→                  </Select>
   347→                </div>
   348→              </CardContent>
   349→            </Card>
   350→          </motion.div>
   351→        </TabsContent>
   352→
   353→        {/* ================================================================= */}
   354→        {/* NOTIFICATIONS TAB                                                 */}
   355→        {/* ================================================================= */}
   356→        <TabsContent value="notifications" className="space-y-6">
   357→          <motion.div variants={itemVariants}>
   358→            <Card>
   359→              <CardHeader>
   360→                <CardTitle className="text-lg flex items-center gap-2">
   361→                  <Bell className="w-5 h-5 text-amber-500" />
   362→                  Notification Preferences
   363→                </CardTitle>
   364→                <CardDescription>Choose which notifications you want to receive</CardDescription>
   365→              </CardHeader>
   366→              <CardContent className="space-y-1">
   367→                {/* Email notifications */}
   368→                <NotificationRow
   369→                  icon={<Mail className="w-4 h-4 text-blue-500" />}
   370→                  bgColor="bg-blue-100 dark:bg-blue-900/30"
   371→                  title="Email Notifications"
   372→                  description="Receive important updates via email"
   373→                  checked={notifPrefs.email}
   374→                  onCheckedChange={(val) => handleNotifToggle('email', val)}
   375→                />
   376→
   377→                <Separator />
   378→
   379→                {/* Attendance alerts */}
   380→                <NotificationRow
   381→                  icon={<ClipboardCheck className="w-4 h-4 text-emerald-500" />}
   382→                  bgColor="bg-emerald-100 dark:bg-emerald-900/30"
   383→                  title="Attendance Alerts"
   384→                  description="Get notified about attendance status changes"
   385→                  checked={notifPrefs.attendance}
   386→                  onCheckedChange={(val) => handleNotifToggle('attendance', val)}
   387→                />
   388→
   389→                <Separator />
   390→
   391→                {/* Assignment reminders */}
   392→                <NotificationRow
   393→                  icon={<FileText className="w-4 h-4 text-orange-500" />}
   394→                  bgColor="bg-orange-100 dark:bg-orange-900/30"
   395→                  title="Assignment Reminders"
   396→                  description="Reminders for upcoming assignment deadlines"
   397→                  checked={notifPrefs.assignments}
   398→                  onCheckedChange={(val) => handleNotifToggle('assignments', val)}
   399→                />
   400→
   401→                <Separator />
   402→
   403→                {/* Notice alerts */}
   404→                <NotificationRow
   405→                  icon={<AlertCircle className="w-4 h-4 text-red-500" />}
   406→                  bgColor="bg-red-100 dark:bg-red-900/30"
   407→                  title="Notice Alerts"
   408→                  description="Alerts for new and urgent notices"
   409→                  checked={notifPrefs.notices}
   410→                  onCheckedChange={(val) => handleNotifToggle('notices', val)}
   411→                />
   412→
   413→                <Separator />
   414→
   415→                {/* Recommendation alerts */}
   416→                <NotificationRow
   417→                  icon={<Lightbulb className="w-4 h-4 text-yellow-500" />}
   418→                  bgColor="bg-yellow-100 dark:bg-yellow-900/30"
   419→                  title="Recommendation Alerts"
   420→                  description="Get personalized academic recommendations"
   421→                  checked={notifPrefs.recommendations}
   422→                  onCheckedChange={(val) => handleNotifToggle('recommendations', val)}
   423→                />
   424→              </CardContent>
   425→            </Card>
   426→          </motion.div>
   427→        </TabsContent>
   428→
   429→        {/* ================================================================= */}
   430→        {/* SECURITY TAB                                                      */}
   431→        {/* ================================================================= */}
   432→        <TabsContent value="security" className="space-y-6">
   433→          {/* Account Info Card */}
   434→          <motion.div variants={itemVariants}>
   435→            <Card>
   436→              <CardHeader>
   437→                <CardTitle className="text-lg flex items-center gap-2">
   438→                  <User className="w-5 h-5 text-emerald-500" />
   439→                  Account Information
   440→                </CardTitle>
   441→                <CardDescription>Your account details (read-only)</CardDescription>
   442→              </CardHeader>
   443→              <CardContent className="space-y-4">
   444→                <div className="grid grid-cols-1 md:grid-cols-3 gap-4">
   445→                  <div className="space-y-1">
   446→                    <Label className="text-xs text-muted-foreground uppercase tracking-wider">Name</Label>
   447→                    <p className="text-sm font-medium">{currentUser.name}</p>
   448→                  </div>
   449→                  <div className="space-y-1">
   450→                    <Label className="text-xs text-muted-foreground uppercase tracking-wider">Email</Label>
   451→                    <p className="text-sm font-medium">{currentUser.email}</p>
   452→                  </div>
   453→                  <div className="space-y-1">
   454→                    <Label className="text-xs text-muted-foreground uppercase tracking-wider">Role</Label>
   455→                    <Badge
   456→                      variant="outline"
   457→                      className="capitalize mt-0.5 bg-emerald-100 text-emerald-800 border-emerald-200 dark:bg-emerald-900/30 dark:text-emerald-400 dark:border-emerald-800"
   458→                    >
   459→                      {currentUser.role}
   460→                    </Badge>
   461→                  </div>
   462→                </div>
   463→
   464→                <Separator />
   465→
   466→                <Button
   467→                  variant="outline"
   468→                  className="gap-2 border-emerald-300 text-emerald-700 hover:bg-emerald-50 dark:border-emerald-700 dark:text-emerald-400 dark:hover:bg-emerald-950/30"
   469→                  onClick={navigateToProfile}
   470→                >
   471→                  <User className="w-4 h-4" />
   472→                  Go to Profile
   473→                  <ChevronRight className="w-4 h-4 ml-auto" />
   474→                </Button>
   475→              </CardContent>
   476→            </Card>
   477→          </motion.div>
   478→
   479→          {/* Change Password Card */}
   480→          <motion.div variants={itemVariants}>
   481→            <Card>
   482→              <CardHeader>
   483→                <CardTitle className="text-lg flex items-center gap-2">
   484→                  <KeyRound className="w-5 h-5 text-amber-500" />
   485→                  Change Password
   486→                </CardTitle>
   487→                <CardDescription>Update your password to keep your account secure</CardDescription>
   488→              </CardHeader>
   489→              <CardContent>
   490→                <form onSubmit={handleChangePassword} className="space-y-4">
   491→                  <div className="space-y-2">
   492→                    <Label htmlFor="settings-current-password">Current Password</Label>
   493→                    <div className="relative">
   494→                      <Input
   495→                        id="settings-current-password"
   496→                        type={showCurrent ? 'text' : 'password'}
   497→                        value={currentPassword}
   498→                        onChange={(e) => setCurrentPassword(e.target.value)}
   499→                        placeholder="Enter current password"
   500→                        required
   501→                      />
   502→                      <Button
   503→                        type="button"
   504→                        variant="ghost"
   505→                        size="icon"
   506→                        className="absolute right-0 top-0 h-full px-3 hover:bg-transparent"
   507→                        onClick={() => setShowCurrent(!showCurrent)}
   508→                      >
   509→                        {showCurrent ? <EyeOff className="w-4 h-4 text-muted-foreground" /> : <Eye className="w-4 h-4 text-muted-foreground" />}
   510→                      </Button>
   511→                    </div>
   512→                  </div>
   513→                  <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
   514→                    <div className="space-y-2">
   515→                      <Label htmlFor="settings-new-password">New Password</Label>
   516→                      <div className="relative">
   517→                        <Input
   518→                          id="settings-new-password"
   519→                          type={showNew ? 'text' : 'password'}
   520→                          value={newPassword}
   521→                          onChange={(e) => setNewPassword(e.target.value)}
   522→                          placeholder="Enter new password"
   523→                          required
   524→                          minLength={6}
   525→                        />
   526→                        <Button
   527→                          type="button"
   528→                          variant="ghost"
   529→                          size="icon"
   530→                          className="absolute right-0 top-0 h-full px-3 hover:bg-transparent"
   531→                          onClick={() => setShowNew(!showNew)}
   532→                        >
   533→                          {showNew ? <EyeOff className="w-4 h-4 text-muted-foreground" /> : <Eye className="w-4 h-4 text-muted-foreground" />}
   534→                        </Button>
   535→                      </div>
   536→                    </div>
   537→                    <div className="space-y-2">
   538→                      <Label htmlFor="settings-confirm-password">Confirm New Password</Label>
   539→                      <Input
   540→                        id="settings-confirm-password"
   541→                        type="password"
   542→                        value={confirmPassword}
   543→                        onChange={(e) => setConfirmPassword(e.target.value)}
   544→                        placeholder="Confirm new password"
   545→                        required
   546→                        minLength={6}
   547→                      />
   548→                    </div>
   549→                  </div>
   550→                  {newPassword && confirmPassword && newPassword !== confirmPassword && (
   551→                    <p className="text-xs text-destructive flex items-center gap-1">
   552→                      <AlertTriangle className="w-3 h-3" /> Passwords do not match
   553→                    </p>

   556→                    <KeyRound className="w-4 h-4 mr-2" />
   557→                    {changingPassword ? 'Changing Password...' : 'Change Password'}

   559→                </form>
   560→              </CardContent>
   561→            </Card>
   562→          </motion.div>
