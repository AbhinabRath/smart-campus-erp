     1→// =============================================================================
     2→// Smart Campus ERP - Attendance Manager Component
     3→// =============================================================================
     4→// Role-based attendance management:
     5→// - Teacher: Create sessions, display QR code, end sessions, view reports
     6→// - Student: Enter QR code to mark attendance, view history & percentage
     7→// - Admin: View attendance reports and analytics
     8→// =============================================================================
     9→
    10→'use client';
    11→
    12→import React, { useEffect, useState, useCallback } from 'react';
    13→import { motion, AnimatePresence } from 'framer-motion';
    14→import {
    15→  ClipboardCheck, QrCode, Play, StopCircle, RefreshCw, CheckCircle, XCircle, Users, Download, BarChart3, Calendar
    16→} from 'lucide-react';
    17→import { Card, CardContent, CardHeader, CardTitle } from '@/components/ui/card';
    18→import { Button } from '@/components/ui/button';
    19→import { Input } from '@/components/ui/input';
    20→import { Label } from '@/components/ui/label';
    21→import { Badge } from '@/components/ui/badge';
    22→import { Select, SelectContent, SelectItem, SelectTrigger, SelectValue } from '@/components/ui/select';
    23→import { Table, TableBody, TableCell, TableHead, TableHeader, TableRow } from '@/components/ui/table';
    24→import { useAppStore } from '@/lib/store';
    25→import api from '@/lib/api';
    26→import { format } from 'date-fns';
    27→import { useToast } from '@/hooks/use-toast';
    28→
    29→interface Subject { id: string; name: string; code: string; }
    30→interface Department { id: string; name: string; code: string; }
    31→interface Session {
    32→  id: string; qrCode: string; isActive: boolean; duration: number; startedAt: string; endedAt: string | null;
    33→  subject: { name: string; code: string }; teacher: { user: { name: string } };
    34→  department: { id?: string; name: string }; semester: number; section: string; _count: { records: number };
    35→  qrDataUrl?: string;
    36→}
    37→interface Record { id: string; markedAt: string; session: { subject: { name: string }; startedAt: string; }; }
    38→
    39→export default function AttendanceManager() {
    40→  const { currentUser } = useAppStore();
    41→  const role = currentUser?.role || 'student';
    42→  const { toast } = useToast();
    43→
    44→  // Teacher state
    45→  const [subjects, setSubjects] = useState<Subject[]>([]);
    46→  const [departments, setDepartments] = useState<Department[]>([]);
    47→  const [sessions, setSessions] = useState<Session[]>([]);
    48→  const [activeSession, setActiveSession] = useState<Session | null>(null);
    49→  const [selectedSubject, setSelectedSubject] = useState('');
    50→  const [selectedDept, setSelectedDept] = useState('');
    51→  const [selectedSemester, setSelectedSemester] = useState('3');
    52→  const [selectedSection, setSelectedSection] = useState('A');
    53→  const [creating, setCreating] = useState(false);
    54→  const [filterDate, setFilterDate] = useState(format(new Date(), 'yyyy-MM-dd'));
    55→
    56→  // Student state
    57→  const [qrInput, setQrInput] = useState('');
    58→  const [marking, setMarking] = useState(false);
    59→  const [myRecords, setMyRecords] = useState<Record[]>([]);
    60→  const [attendancePercent, setAttendancePercent] = useState({ percentage: 0, totalSessions: 0, attended: 0 });
    61→
    62→  // Admin state
    63→  const [adminFilterDept, setAdminFilterDept] = useState('all');
    64→  const [adminFilterSemester, setAdminFilterSemester] = useState('all');
    65→  const [adminLoading, setAdminLoading] = useState(true);
    66→
    67→  // Load dropdown data for teachers
    68→  const loadTeacherData = useCallback(async () => {
    69→    try {
    70→      const [subRes, deptRes, sessRes] = await Promise.all([
    71→        api.get('/subjects'),
    72→        api.get('/departments'),
    73→        api.get('/attendance/sessions'),
    74→      ]);
    75→      setSubjects(subRes.data.data || []);
    76→      setDepartments(deptRes.data.data || []);
    77→      const allSessions = sessRes.data.data || [];
    78→      setSessions(allSessions);
    79→      const active = allSessions.find((s: Session) => s.isActive);
    80→      if (active) setActiveSession(active);
    81→    } catch { /* ignore */ }
    82→  }, []);
    83→
    84→  // Load student data
    85→  const loadStudentData = useCallback(async () => {
    86→    try {
    87→      const [histRes, pctRes] = await Promise.all([
    88→        api.get('/attendance/my-attendance'),
    89→        api.get('/attendance/percentage'),
    90→      ]);
    91→      setMyRecords(histRes.data.data || []);
    92→      setAttendancePercent(pctRes.data.data || { percentage: 0, totalSessions: 0, attended: 0 });
    93→    } catch { /* ignore */ }
    94→  }, []);
    95→
    96→  // Admin: load data with loading state
    97→  const loadAdminData = useCallback(async () => {
    98→    setAdminLoading(true);
    99→    try {
   100→      const [sessRes, deptRes] = await Promise.all([
   101→        api.get('/attendance/sessions'),
   102→        api.get('/departments'),
   103→      ]);
   104→      const allSessions = sessRes.data.data || [];
   105→      setSessions(allSessions);
   106→      setDepartments(deptRes.data.data || []);
   107→    } catch { /* ignore */ }
   108→    setAdminLoading(false);
   109→  }, []);
   110→
   111→  useEffect(() => {
   112→    if (role === 'teacher') loadTeacherData();
   113→    if (role === 'admin') loadAdminData();
   114→    if (role === 'student') loadStudentData();
   115→  }, [role, loadTeacherData, loadStudentData, loadAdminData]);
   116→
   117→  // Teacher: Create attendance session
   118→  const createSession = async () => {
   119→    if (!selectedSubject || !selectedDept) {
   120→      toast({ title: 'Error', description: 'Please select subject and department', variant: 'destructive' });
   121→      return;
   122→    }
   123→    setCreating(true);
   124→    try {
   125→      const res = await api.post('/attendance/sessions', {
   126→        subjectId: selectedSubject,
   127→        departmentId: selectedDept,
   128→        semester: parseInt(selectedSemester),
   129→        section: selectedSection,
   130→      });
   131→      const newSession = res.data.data;
   132→      setActiveSession(newSession);
   133→      setSessions((prev) => [newSession, ...prev]);
   134→      toast({ title: 'Session Created', description: 'Attendance session is now active' });
   135→    } catch (err: unknown) {
   136→      const axiosErr = err as { response?: { data?: { message?: string } } };
   137→      toast({ title: 'Error', description: axiosErr.response?.data?.message || 'Failed to create session', variant: 'destructive' });
   138→    } finally {
   139→      setCreating(false);
   140→    }
   141→  };
   142→
   143→  // Teacher: End attendance session
   144→  const endSession = async (sessionId: string) => {
   145→    try {
   146→      await api.post(`/attendance/sessions/${sessionId}/end`);
   147→      setActiveSession(null);
   148→      setSessions((prev) => prev.map((s) => s.id === sessionId ? { ...s, isActive: false } : s));
   149→      toast({ title: 'Session Ended', description: 'Attendance session has been closed' });
   150→    } catch {
   151→      toast({ title: 'Error', description: 'Failed to end session', variant: 'destructive' });
   152→    }
   153→  };
   154→
   155→  // Export CSV helper
   156→  const exportSessionsCSV = (sessionsData: Session[], filename: string) => {
   157→    if (sessionsData.length === 0) {
   158→      toast({ title: 'No Data', description: 'No sessions to export', variant: 'destructive' });
   159→      return;
   160→    }
   161→    const headers = ['Subject', 'Teacher', 'Department', 'Semester', 'Section', 'Started', 'Present Count', 'Status'];
   162→    const rows = sessionsData.map((s) => [
   163→      s.subject?.name || 'N/A',
   164→      s.teacher?.user?.name || 'N/A',
   165→      s.department?.name || 'N/A',
   166→      s.semester?.toString() || '',
   167→      s.section || '',
   168→      s.startedAt ? format(new Date(s.startedAt), 'MMM d, yyyy h:mm a') : '',
   169→      (s._count?.records || 0).toString(),
   170→      s.isActive ? 'Live' : 'Ended',
   171→    ]);
   172→    const csvContent = [headers, ...rows].map((row) => row.map((cell) => `"${cell.replace(/"/g, '""')}"`).join(',')).join('\n');
   173→    const blob = new Blob([csvContent], { type: 'text/csv;charset=utf-8;' });
   174→    const url = URL.createObjectURL(blob);
   175→    const link = document.createElement('a');
   176→    link.href = url;
   177→    link.download = filename;
   178→    link.click();
   179→    URL.revokeObjectURL(url);
   180→    toast({ title: 'Exported', description: 'CSV file downloaded successfully' });
   181→  };
   182→
   183→  // Student: Mark attendance via QR code
   184→  const markAttendance = async () => {
   185→    if (!qrInput.trim()) return;
   186→    setMarking(true);
   187→    try {
   188→      await api.post('/attendance/mark', { qrCode: qrInput.trim() });
   189→      toast({ title: 'Success', description: 'Attendance marked successfully!' });
   190→      setQrInput('');
   191→      loadStudentData();
   192→    } catch (err: unknown) {
   193→      const axiosErr = err as { response?: { data?: { message?: string } } };
   194→      toast({ title: 'Error', description: axiosErr.response?.data?.message || 'Failed to mark attendance', variant: 'destructive' });
   195→    } finally {
   196→      setMarking(false);
   197→    }
   198→  };
   199→
   200→  // =========================================================================
   201→  // Teacher View
   202→  // =========================================================================
   203→  if (role === 'teacher') {
   204→    return (
   205→      <div className="space-y-6">
   206→        <div className="flex items-center justify-between">
   207→          <h2 className="text-2xl font-bold tracking-tight">Attendance Management</h2>
   208→          <div className="flex items-center gap-2">
   209→            <Input type="date" value={filterDate} onChange={(e) => setFilterDate(e.target.value)} className="h-8 text-xs w-[150px]" />
   210→          </div>
   211→        </div>
   212→
   213→        <div className="grid grid-cols-1 lg:grid-cols-2 gap-6">
   214→          {/* Create Session */}
   215→          <motion.div initial={{ opacity: 0, y: 20 }} animate={{ opacity: 1, y: 0 }}>
   216→            <Card className="overflow-hidden">
   217→              <div className="h-1.5 bg-gradient-to-r from-emerald-500 to-teal-500" />
   218→              <CardHeader><CardTitle className="text-lg flex items-center gap-2"><Play className="w-5 h-5 text-emerald-600" /> Create Session</CardTitle></CardHeader>
   219→              <CardContent className="space-y-4">
   220→                <div className="space-y-2">
   221→                  <Label>Subject</Label>
   222→                  <Select value={selectedSubject} onValueChange={setSelectedSubject}>
   223→                    <SelectTrigger><SelectValue placeholder="Select subject" /></SelectTrigger>
   224→                    <SelectContent>
   225→                      {subjects.map((s) => <SelectItem key={s.id} value={s.id}>{s.name} ({s.code})</SelectItem>)}
   226→                    </SelectContent>
   227→                  </Select>
   228→                </div>
   229→                <div className="space-y-2">
   230→                  <Label>Department</Label>
   231→                  <Select value={selectedDept} onValueChange={setSelectedDept}>
   232→                    <SelectTrigger><SelectValue placeholder="Select department" /></SelectTrigger>
   233→                    <SelectContent>
   234→                      {departments.map((d) => <SelectItem key={d.id} value={d.id}>{d.name} ({d.code})</SelectItem>)}
   235→                    </SelectContent>
   236→                  </Select>
   237→                </div>
   238→                <div className="grid grid-cols-2 gap-4">
   239→                  <div className="space-y-2">
   240→                    <Label>Semester</Label>
   241→                    <Select value={selectedSemester} onValueChange={setSelectedSemester}>
   242→                      <SelectTrigger><SelectValue /></SelectTrigger>
   243→                      <SelectContent>
   244→                        {[1,2,3,4,5,6,7,8].map((s) => <SelectItem key={s} value={s.toString()}>{s}</SelectItem>)}
   245→                      </SelectContent>
   246→                    </Select>
   247→                  </div>
   248→                  <div className="space-y-2">
   249→                    <Label>Section</Label>
   250→                    <Select value={selectedSection} onValueChange={setSelectedSection}>
   251→                      <SelectTrigger><SelectValue /></SelectTrigger>
   252→                      <SelectContent>
   253→                        {['A','B','C'].map((s) => <SelectItem key={s} value={s}>{s}</SelectItem>)}
   254→                      </SelectContent>
   255→                    </Select>
   256→                  </div>
   257→                </div>
   258→                <Button onClick={createSession} disabled={creating || !!activeSession} className="w-full bg-emerald-700 hover:bg-emerald-800">
   259→                  {creating ? 'Creating...' : activeSession ? 'Session Active' : 'Create Session'}
   260→                </Button>
   261→              </CardContent>
   262→            </Card>
   263→          </motion.div>
   264→
   265→          {/* Active Session / QR Code */}
   266→          <motion.div initial={{ opacity: 0, y: 20 }} animate={{ opacity: 1, y: 0 }} transition={{ delay: 0.1 }}>
   267→            <Card className="overflow-hidden">
   268→              <div className="h-1.5 bg-gradient-to-r from-emerald-500 to-teal-500" />
   269→              <CardHeader><CardTitle className="text-lg flex items-center gap-2"><QrCode className="w-5 h-5 text-emerald-600" /> Active Session</CardTitle></CardHeader>
   270→              <CardContent>
   271→                {activeSession ? (
   272→                  <div className="space-y-4 text-center">
   273→                    <Badge className="bg-emerald-600 mb-2 flex items-center gap-1.5 mx-auto w-fit">
   274→                      <span className="relative flex h-2 w-2">
   275→                        <span className="animate-ping absolute inline-flex h-full w-full rounded-full bg-white opacity-75" />
   276→                        <span className="relative inline-flex rounded-full h-2 w-2 bg-white" />
   277→                      </span>
   278→                      Live
   279→                    </Badge>
   280→                    <p className="font-medium">{activeSession.subject?.name || 'Session'}</p>
   281→                    <p className="text-sm text-muted-foreground">
   282→                      Started {format(new Date(activeSession.startedAt), 'h:mm a')}
   283→                    </p>
   284→                    {activeSession.qrDataUrl && (
   285→                      <div className="flex justify-center">
   286→                        <div className="relative p-4 border-2 border-emerald-200 dark:border-emerald-800 rounded-xl bg-white">
   287→                          {/* Corner markers */}
   288→                          <div className="absolute top-1 left-1 w-4 h-4 border-t-2 border-l-2 border-emerald-500 rounded-tl" />
   289→                          <div className="absolute top-1 right-1 w-4 h-4 border-t-2 border-r-2 border-emerald-500 rounded-tr" />
   290→                          <div className="absolute bottom-1 left-1 w-4 h-4 border-b-2 border-l-2 border-emerald-500 rounded-bl" />
   291→                          <div className="absolute bottom-1 right-1 w-4 h-4 border-b-2 border-r-2 border-emerald-500 rounded-br" />
   292→                          <img src={activeSession.qrDataUrl} alt="QR Code" className="w-44 h-44 rounded-lg" />
   293→                        </div>
   294→                      </div>
   295→                    )}
   296→                    <p className="text-xs text-muted-foreground">QR Token: {activeSession.qrCode}</p>
   297→                    <Button variant="destructive" onClick={() => endSession(activeSession.id)} className="w-full">
   298→                      <StopCircle className="w-4 h-4 mr-2" /> End Session
   299→                    </Button>
   300→                  </div>
   301→                ) : (
   302→                  <div className="text-center py-8 text-muted-foreground">
   303→                    <QrCode className="w-12 h-12 mx-auto mb-3 opacity-30" />
   304→                    <p>No active session</p>
   305→                    <p className="text-sm">Create a session to generate a QR code</p>
   306→                  </div>
   307→                )}
   308→              </CardContent>
   309→            </Card>
   310→          </motion.div>
   311→        </div>
   312→
   313→        {/* Session History */}
   314→        <motion.div initial={{ opacity: 0, y: 20 }} animate={{ opacity: 1, y: 0 }} transition={{ delay: 0.2 }}>
   315→          <Card className="overflow-hidden">
   316→            <div className="h-1 bg-gradient-to-r from-emerald-500 via-teal-400 to-emerald-500" />
   317→            <CardHeader className="flex flex-row items-center justify-between">
   318→              <CardTitle className="text-lg flex items-center gap-2"><Calendar className="w-5 h-5 text-emerald-600" /> Session History</CardTitle>
   319→              <div className="flex items-center gap-1">
   320→                <Button variant="ghost" size="icon" onClick={() => exportSessionsCSV(sessions, 'attendance-sessions.csv')} title="Export CSV">
   321→                  <Download className="w-4 h-4" />
   322→                </Button>
   323→                <Button variant="ghost" size="icon" onClick={loadTeacherData}><RefreshCw className="w-4 h-4" /></Button>
   324→              </div>
   325→            </CardHeader>
   326→            <CardContent className="max-h-96 overflow-y-auto custom-scrollbar">
   327→              {sessions.length === 0 ? (
   328→                <p className="text-center text-muted-foreground py-4">No sessions yet</p>
   329→              ) : (
   330→                <Table>
   331→                  <TableHeader>
   332→                    <TableRow>
   333→                      <TableHead>Subject</TableHead>
   334→                      <TableHead>Sem/Sec</TableHead>
   335→                      <TableHead>Started</TableHead>
   336→                      <TableHead>Present</TableHead>
   337→                      <TableHead>Status</TableHead>
   338→                      <TableHead>Actions</TableHead>
   339→                    </TableRow>
   340→                  </TableHeader>
   341→                  <TableBody>
   342→                    {sessions.map((s) => (
   343→                      <TableRow key={s.id}>
   344→                        <TableCell className="font-medium">{s.subject?.name || 'N/A'}</TableCell>
   345→                        <TableCell>{s.semester}/{s.section}</TableCell>
   346→                        <TableCell>{format(new Date(s.startedAt), 'MMM d, h:mm a')}</TableCell>
   347→                        <TableCell>{s._count?.records || 0}</TableCell>
   348→                        <TableCell>
   349→                          <Badge variant={s.isActive ? 'default' : 'secondary'} className={s.isActive ? 'bg-emerald-600 flex items-center gap-1.5' : ''}>
   350→                            {s.isActive && (
   351→                              <span className="relative flex h-1.5 w-1.5">
   352→                                <span className="animate-ping absolute inline-flex h-full w-full rounded-full bg-white opacity-75" />
   353→                                <span className="relative inline-flex rounded-full h-1.5 w-1.5 bg-white" />
   354→                              </span>
   355→                            )}
   356→                            {s.isActive ? 'Live' : 'Ended'}
   357→                          </Badge>
   358→                        </TableCell>
   359→                        <TableCell>
   360→                          {s.isActive && (
   361→                            <Button size="sm" variant="outline" onClick={() => endSession(s.id)}>End</Button>
   362→                          )}
   363→                        </TableCell>
   364→                      </TableRow>
   365→                    ))}
   366→                  </TableBody>
   367→                </Table>
   368→              )}
   369→            </CardContent>
   370→          </Card>
   371→        </motion.div>
   372→      </div>
   373→    );
   374→  }
   375→
   376→  // =========================================================================
   377→  // Student View
   378→  // =========================================================================
   379→  if (role === 'student') {
   380→    return (
   381→      <div className="space-y-6">
   382→        <h2 className="text-2xl font-bold tracking-tight">My Attendance</h2>
   383→
   384→        <div className="grid grid-cols-1 sm:grid-cols-3 gap-4">
   385→          <motion.div initial={{ opacity: 0, y: 20 }} animate={{ opacity: 1, y: 0 }}>
   386→            <Card className="overflow-hidden">
   387→              <div className="h-1 bg-gradient-to-r from-emerald-500 to-teal-400" />
   388→              <CardContent className="p-5 text-center">
   389→                <p className="text-sm text-muted-foreground">Attendance %</p>
   390→                <p className={`text-3xl font-bold ${attendancePercent.percentage >= 75 ? 'text-emerald-600' : 'text-amber-600'}`}>
   391→                  {attendancePercent.percentage}%
   392→                </p>
   393→              </CardContent>
   394→            </Card>
   395→          </motion.div>
   396→          <motion.div initial={{ opacity: 0, y: 20 }} animate={{ opacity: 1, y: 0 }} transition={{ delay: 0.1 }}>
   397→            <Card className="overflow-hidden">
   398→              <div className="h-1 bg-gradient-to-r from-sky-500 to-blue-400" />
   399→              <CardContent className="p-5 text-center">
   400→                <p className="text-sm text-muted-foreground">Total Sessions</p>
   401→                <p className="text-3xl font-bold">{attendancePercent.totalSessions}</p>
   402→              </CardContent>
   403→            </Card>
   404→          </motion.div>
   405→          <motion.div initial={{ opacity: 0, y: 20 }} animate={{ opacity: 1, y: 0 }} transition={{ delay: 0.2 }}>
   406→            <Card className="overflow-hidden">
   407→              <div className="h-1 bg-gradient-to-r from-emerald-500 to-green-400" />
   408→              <CardContent className="p-5 text-center">
   409→                <p className="text-sm text-muted-foreground">Attended</p>
   410→                <p className="text-3xl font-bold text-emerald-600">{attendancePercent.attended}</p>
   411→              </CardContent>
   412→            </Card>
   413→          </motion.div>
   414→        </div>
   415→
   416→        {/* QR Code Input */}
   417→        <motion.div initial={{ opacity: 0, y: 20 }} animate={{ opacity: 1, y: 0 }} transition={{ delay: 0.3 }}>
   418→          <Card className="overflow-hidden">
   419→            <div className="h-1 bg-gradient-to-r from-emerald-500 to-teal-500" />
   420→            <CardHeader><CardTitle className="text-lg flex items-center gap-2"><QrCode className="w-5 h-5 text-emerald-600" /> Mark Attendance</CardTitle></CardHeader>
   421→            <CardContent>
   422→              <div className="flex gap-3">
   423→                <Input
   424→                  placeholder="Enter QR code token..."
   425→                  value={qrInput}
   426→                  onChange={(e) => setQrInput(e.target.value)}
   427→                  onKeyDown={(e) => e.key === 'Enter' && markAttendance()}
   428→                  className="flex-1"
   429→                />
   430→                <Button onClick={markAttendance} disabled={marking || !qrInput.trim()} className="bg-emerald-700 hover:bg-emerald-800">
   431→                  {marking ? 'Marking...' : <><CheckCircle className="w-4 h-4 mr-2" /> Mark</>}
   432→                </Button>
   433→              </div>
   434→              <p className="text-xs text-muted-foreground mt-2">Enter the QR code token provided by your teacher to mark attendance</p>
   435→            </CardContent>
   436→          </Card>
   437→        </motion.div>
   438→
   439→        {/* Attendance History */}
   440→        <motion.div initial={{ opacity: 0, y: 20 }} animate={{ opacity: 1, y: 0 }} transition={{ delay: 0.4 }}>
   441→          <Card className="overflow-hidden">
   442→            <div className="h-1 bg-gradient-to-r from-emerald-500 via-teal-400 to-emerald-500" />
   443→            <CardHeader className="flex flex-row items-center justify-between">
   444→              <CardTitle className="text-lg flex items-center gap-2"><Calendar className="w-5 h-5 text-emerald-600" /> Attendance History</CardTitle>
   445→              <Button variant="ghost" size="icon" onClick={loadStudentData}><RefreshCw className="w-4 h-4" /></Button>
   446→            </CardHeader>
   447→            <CardContent className="max-h-96 overflow-y-auto custom-scrollbar">
   448→              {myRecords.length === 0 ? (
   449→                <div className="text-center py-8 text-muted-foreground">
   450→                  <ClipboardCheck className="w-12 h-12 mx-auto mb-3 opacity-30" />
   451→                  <p>No attendance records yet</p>
   452→                </div>
   453→              ) : (
   454→                <Table>
   455→                  <TableHeader>
   456→                    <TableRow>
   457→                      <TableHead>Subject</TableHead>
   458→                      <TableHead>Date</TableHead>
