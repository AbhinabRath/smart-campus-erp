    67→// Circular Progress Indicator for Attendance
    68→function CircularProgress({ value, size = 140, strokeWidth = 10 }: { value: number; size?: number; strokeWidth?: number }) {
    69→  const radius = (size - strokeWidth) / 2;
    70→  const circumference = 2 * Math.PI * radius;
    71→  const offset = circumference - (value / 100) * circumference;
    72→  const color = value >= 75 ? '#10b981' : value >= 50 ? '#f59e0b' : '#ef4444';
    73→
    74→  return (
    75→    <div className="relative inline-flex items-center justify-center">
    76→      <svg width={size} height={size} className="-rotate-90">
    77→        <circle
    78→          cx={size / 2}
    79→          cy={size / 2}
    80→          r={radius}
    81→          stroke="currentColor"
    82→          strokeWidth={strokeWidth}
    83→          fill="none"
    84→          className="text-muted/30"
    85→        />
    86→        <motion.circle
    87→          cx={size / 2}
    88→          cy={size / 2}
    89→          r={radius}
    90→          stroke={color}
    91→          strokeWidth={strokeWidth}
    92→          fill="none"
    93→          strokeLinecap="round"
    94→          strokeDasharray={circumference}
    95→          initial={{ strokeDashoffset: circumference }}
    96→          animate={{ strokeDashoffset: offset }}
    97→          transition={{ duration: 1.5, ease: 'easeOut' }}
    98→        />
    99→      </svg>
   100→      <div className="absolute inset-0 flex flex-col items-center justify-center">
   101→        <span className="text-3xl font-bold" style={{ color }}>{value}%</span>
   102→        <span className="text-xs text-muted-foreground">Attendance</span>
   103→      </div>
   104→    </div>
   105→  );
   106→}

   108→// Countdown display for deadline
   109→function DeadlineCountdown({ deadline }: { deadline: string }) {
   110→  const now = new Date();
   111→  const deadlineDate = new Date(deadline);
   112→  const hoursLeft = differenceInHours(deadlineDate, now);
   113→  const daysLeft = differenceInDays(deadlineDate, now);
   114→  const isOverdue = hoursLeft < 0;
   115→  const isUrgent = !isOverdue && hoursLeft < 48;
   116→
   117→  return (
   118→    <div className={`flex items-center gap-1 text-xs font-medium ${isOverdue ? 'text-destructive' : isUrgent ? 'text-amber-600 dark:text-amber-400' : 'text-muted-foreground'}`}>
   119→      {isOverdue ? (
   120→        <>
   121→          <AlertTriangle className="w-3 h-3" />
   122→          Overdue
   123→        </>
   124→      ) : daysLeft > 0 ? (
   125→        <>
   126→          <Clock className="w-3 h-3" />
   127→          {daysLeft}d {hoursLeft % 24}h left
   128→        </>
   129→      ) : (
   130→        <>
   131→          <Clock className="w-3 h-3" />
   132→          {hoursLeft}h left
   133→        </>
   134→      )}
   135→    </div>
   136→  );
   137→}

   139→// Simplified 10-point GPA scale converter
   140→function marksToGPA(percentage: number): number {
   141→  if (percentage >= 90) return 10;
   142→  if (percentage >= 80) return 9;
   143→  if (percentage >= 70) return 8;
   144→  if (percentage >= 60) return 7;
   145→  if (percentage >= 50) return 6;
   146→  if (percentage >= 45) return 5;
   147→  if (percentage >= 40) return 4;
   148→  return 0;
   149→}
   150→
   151→function gradeLabel(percentage: number): string {
   152→  if (percentage >= 90) return 'A+';
   153→  if (percentage >= 80) return 'A';
   154→  if (percentage >= 70) return 'B+';
   155→  if (percentage >= 60) return 'B';
   156→  if (percentage >= 50) return 'C';
   157→  if (percentage >= 45) return 'D';
   158→  if (percentage >= 40) return 'E';
   159→  return 'F';
   160→}
   161→
   162→function gradeColor(grade: string): string {
   163→  switch (grade) {
   164→    case 'A+': return 'bg-emerald-100 text-emerald-800 dark:bg-emerald-900/30 dark:text-emerald-400';
   165→    case 'A': return 'bg-emerald-100 text-emerald-700 dark:bg-emerald-900/30 dark:text-emerald-400';
   166→    case 'B+': return 'bg-sky-100 text-sky-800 dark:bg-sky-900/30 dark:text-sky-400';
   167→    case 'B': return 'bg-sky-100 text-sky-700 dark:bg-sky-900/30 dark:text-sky-400';
   168→    case 'C': return 'bg-amber-100 text-amber-800 dark:bg-amber-900/30 dark:text-amber-400';
   169→    case 'D': return 'bg-orange-100 text-orange-800 dark:bg-orange-900/30 dark:text-orange-400';
   170→    case 'E': return 'bg-red-100 text-red-800 dark:bg-red-900/30 dark:text-red-400';
   171→    case 'F': return 'bg-red-200 text-red-900 dark:bg-red-900/40 dark:text-red-300';
   172→    default: return 'bg-muted text-muted-foreground';
   173→  }
   174→}
